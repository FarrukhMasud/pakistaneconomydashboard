import {
  sourceNumber, publicationStatus, fiscalMonth, fiscalWindow, monthNumber,
  sourceEvidence, compareEquivalent, consecutiveMonths, sumComplete,
} from './ingestion-validation.mjs';
import { parseFiscalYear } from './sheet-utils.mjs';

function nearest(items, x, maxDistance) {
  const sorted = items.map(item => ({ item, distance: Math.abs(item.x - x) })).sort((a, b) => a.distance - b.distance);
  if (!sorted.length || sorted[0].distance > maxDistance || sorted[0].distance === sorted[1]?.distance) {
    throw new Error(`Services headline: missing or ambiguous header at x=${x}`);
  }
  return sorted[0].item;
}

function rowCells(items, labelPattern) {
  const labels = items.filter(item => labelPattern.test(String(item.text ?? '').trim())).sort((a, b) => a.page - b.page || a.y - b.y);
  const label = labels[0];
  if (!label) throw new Error(`Could not locate exports-of-services headline row ${labelPattern}`);
  return {
    label,
    cells: items.filter(item => item !== label && item.page === label.page && item.x > label.x && Math.abs(item.y - label.y) < 0.12).sort((a, b) => a.x - b.x),
  };
}

function parseColumns(items, row) {
  const headers = items.filter(item => item.page === row.label.page && item.y < Math.min(row.label.y, 7.5));
  const years = headers.filter(item => /^FY\s*\d{2,4}(?:\s*\([PRF]\))?$/i.test(String(item.text).trim()));
  const periods = headers.filter(item => /^(?:[A-Z][a-z]{2,8})(?:[-\u2013][A-Z][a-z]{2,8})?$/.test(String(item.text).trim()))
    .filter(item => !/^(?:Items|Million)$/i.test(item.text));
  const markers = headers.filter(item => /^[PRF]$/.test(String(item.text).trim()));
  if (!years.length || !periods.length) throw new Error('Services headline is missing period/FY headers');
  const columns = row.cells.map(cell => {
    const yearHeader = nearest(years, cell.x, 1.3);
    // Resolve against the FY anchor, not the right-aligned value. A printed
    // merged cumulative header spans two FY columns and may be >2 units from
    // the second numeric cell.
    const periodHeader = nearest(periods, yearHeader.x, 1.8);
    const period = periodHeader.text.trim().replace('\u2013', '-');
    const fiscalYear = parseFiscalYear(yearHeader.text);
    if (fiscalYear === null) throw new Error(`Invalid services fiscal year: ${yearHeader.text}`);
    const marker = markers.filter(item => nearest(years, item.x, 1.3) === yearHeader).map(item => item.text).join(' ');
    const window = /[-\u2013]/.test(period) ? fiscalWindow(period, fiscalYear) : null;
    if (!window) monthNumber(period);
    return {
      value: sourceNumber(cell.text, { context: `services PDF page ${cell.page}, x=${cell.x}, y=${cell.y}` }),
      fiscalYear, period, month: window ? null : fiscalMonth(period, fiscalYear),
      window, x: cell.x, status: publicationStatus(marker || yearHeader.text), sourceStatus: marker || yearHeader.text,
      locator: {
        page: cell.page, rowLabel: row.label.text, x: cell.x, y: cell.y, rawValue: cell.text,
        periodHeader: { x: periodHeader.x, y: periodHeader.y, text: periodHeader.text },
        fiscalYearHeader: { x: yearHeader.x, y: yearHeader.y, text: yearHeader.text },
      },
    };
  });
  if (new Set(columns.map(column => column.locator.fiscalYearHeader.x)).size !== columns.length) {
    throw new Error('Services headline contains duplicate/ambiguous value columns');
  }
  return columns;
}

/** Parse SBP's headline table, retaining its printed periods and PDF locators. */
export function parseServicesHeadline(items, { receipt = null } = {}) {
  const itColumns = parseColumns(items, rowCells(items, /^9\.\s*Telecommunications,\s*Computer,\s*and Information Services$/i));
  const totalColumns = parseColumns(items, rowCells(items, /^2\.\s*Exports of Services$/i));
  const unitHeader = items.find(item => /\(\s*Million\s+US\s*\$\s*\)/i.test(item.text));
  if (!unitHeader) throw new Error('Services headline is missing its printed Million US$ unit');
  const currentFiscalYear = Math.max(...itColumns.map(column => column.fiscalYear));
  const current = itColumns.filter(column => column.fiscalYear === currentFiscalYear).sort((a, b) => a.x - b.x);
  const prior = itColumns.filter(column => column.fiscalYear === currentFiscalYear - 1).sort((a, b) => a.x - b.x);
  const cumulative = current.filter(column => column.window).sort((a, b) => a.window.end.localeCompare(b.window.end));
  const sharedPeriods = current.filter(column => prior.some(other => other.locator.periodHeader.x === column.locator.periodHeader.x));
  const currentFytd = cumulative.at(-1) ?? (sharedPeriods.length === 1 ? sharedPeriods[0] : null);
  const priorCandidates = prior.filter(column => column.period === currentFytd?.period && column.locator.periodHeader.x === currentFytd.locator.periodHeader.x);
  const priorFytd = priorCandidates.length === 1 ? priorCandidates[0] : null;
  const monthly = current.filter(column => column.month && column !== currentFytd).sort((a, b) => a.month.localeCompare(b.month));
  const latest = monthly.at(-1);
  const previous = monthly.at(-2) ?? null;
  const yearAgo = itColumns.find(column => column.month === (latest && `${Number(latest.month.slice(0, 4)) - 1}${latest.month.slice(4)}`) && column !== priorFytd);
  const totalLatest = totalColumns.find(column => column.locator.fiscalYearHeader.x === latest?.locator.fiscalYearHeader.x);
  if (!latest || !yearAgo || !currentFytd || !priorFytd || !totalLatest) {
    throw new Error('Services headline table is missing required monthly or fiscal-year columns');
  }
  const currentWindow = fiscalWindow(currentFytd.period, currentFiscalYear);
  const priorWindow = fiscalWindow(priorFytd.period, currentFiscalYear - 1);
  if (currentFytd.period !== priorFytd.period || currentWindow.end !== latest.month) {
    throw new Error('Services headline cumulative coverage does not match latest/comparable period');
  }
  const locatorColumns = { latest, prev: previous, yearAgo, fytd: currentFytd, fytdPrior: priorFytd, totalServicesLatest: totalLatest };
  const evidence = Object.fromEntries(Object.entries(locatorColumns).filter(([, column]) => column).map(([field, column]) => [
    field, receipt ? sourceEvidence(receipt, column.locator, { rawValue: column.locator.rawValue, sourceUnit: 'US$ million', unit: 'US$ million', scale: 1 }) : { locator: column.locator },
  ]));
  const reconciliation = [];
  if (monthly[0]?.month === currentWindow.start && monthly.at(-1)?.month === currentWindow.end) {
    consecutiveMonths(monthly.map(column => ({ date: column.month })));
    reconciliation.push(compareEquivalent('Headline FYTD = all printed months in the same window', {
      ...currentWindow, basis: 'BOP-IT-credit', unit: 'US$ million', value: currentFytd.value, evidence: evidence.fytd,
    }, {
      ...currentWindow, basis: 'BOP-IT-credit', unit: 'US$ million', value: sumComplete(monthly.map(column => column.value)),
      evidence: { derivation: 'sum of printed monthly IT columns in exact FYTD coverage', inputs: monthly.map(column => receipt ? sourceEvidence(receipt, column.locator) : { locator: column.locator }) },
    }));
  }
  return {
    latestMonth: latest.month, latest: latest.value,
    prevMonth: previous?.month ?? null, prev: previous?.value ?? null,
    yearAgoMonth: yearAgo.month, yearAgo: yearAgo.value,
    fytd: currentFytd.value, fytdPrior: priorFytd.value,
    fytdLabel: `${currentFytd.period} FY${String(currentFiscalYear).slice(-2)}`,
    fytdPriorLabel: `${priorFytd.period} FY${String(currentFiscalYear - 1).slice(-2)}`,
    totalServicesLatest: totalLatest.value, fiscalYear: currentFiscalYear,
    period: currentWindow, priorPeriod: priorWindow, unit: 'US$ million',
    status: latest.status, statuses: Object.fromEntries(Object.entries(locatorColumns).map(([field, column]) => [field, column?.status ?? 'not-stated'])),
    sourceStatuses: Object.fromEntries(Object.entries(locatorColumns).map(([field, column]) => [field, column?.sourceStatus ?? null])),
    evidence, reconciliation,
  };
}
