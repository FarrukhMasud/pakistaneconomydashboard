  Weekly: 21,
  Monthly: 65,

function observationAgeDays(value, now = new Date()) {
  const end = /^\d{4}$/.test(value || '') ? `${value}-12-31` : periodEndDate(value);
  if (!end) return null;
  const date = new Date(`${end}T00:00:00Z`);
  if (Number.isNaN(date.getTime())) return null;
  return Math.floor((now.getTime() - date.getTime()) / 86_400_000);
}

function explicitPublicationDate(data) {
  return data.publicationDate
    || data.sourcePublicationDate
    || data.publishedAt
    || null;
}

function verificationDate(data) {
  return data.verificationDate
    || data.lastChecked
    || data.lastVerified
    || null;
}

function fiscalPeriodEndDate(value) {
  const match = String(value || '').match(/jul(?:y)?[-–\s]+([a-z]+).*?fy\s*(\d{2,4})/i);
  if (!match) return null;

  const months = {
    jan: 1, feb: 2, mar: 3, apr: 4, may: 5, jun: 6,
    jul: 7, aug: 8, sep: 9, oct: 10, nov: 11, dec: 12,
  };
  const month = months[match[1].slice(0, 3).toLowerCase()];
  if (!month) return null;

  const rawFy = Number(match[2]);
  const fiscalYear = rawFy < 100 ? 2000 + rawFy : rawFy;
  const calendarYear = month >= 7 ? fiscalYear - 1 : fiscalYear;
  const end = new Date(Date.UTC(calendarYear, month, 0));
  return end.toISOString().slice(0, 10);
}

function fiscalYearEndDate(value) {
  const match = String(value || '').match(/fy\s*(\d{2,4})/i);
  if (!match) return null;
  const rawFy = Number(match[1]);
  const fiscalYear = rawFy < 100 ? 2000 + rawFy : rawFy;
  return `${fiscalYear}-06-30`;
}

function latestNumeric(series) {
  return [...(series?.data || [])].reverse().find(row => Number.isFinite(row.value)) || null;
}

function nestedSeries(dataset, data, now) {
  const results = [];
  const add = (id, label, observationDate, evidence, cadence = dataset.cadence, publicationStatus = 'not-stated') => {
    const age = observationAgeDays(observationDate, now);
    const limit = MAX_AGE_DAYS[cadence];
    results.push({
      id, label, observationDate: observationDate || null,
      verificationDate: evidence?.retrievedAt || null,
      sourceUrl: evidence?.sourceUrl || null,
      publicationStatus, evidence: evidence || null,
      status: !observationDate ? 'missing' : age != null && limit != null && age > limit ? 'overdue' : 'latest-available',
    });
  };
  const walk = (value, prefix = '') => {
    if (!value || typeof value !== 'object' || Array.isArray(value)) return;
    for (const [key, item] of Object.entries(value)) {
      if (item && typeof item === 'object' && Array.isArray(item.data)) {
        const latest = [...item.data].filter(row => row.value != null).at(-1);
        add(`${prefix}${key}`, item.name || key, latest?.observationDate || latest?.date || fiscalYearEndDate(latest?.fy), latest?.evidence || item.evidence, dataset.cadence, latest?.status);
      } else if (key === 'publicFinance') walk(item, `${key}.`);
    }
  };
  walk(data);
  const fields = dataset.id === 'remittances'
    ? ['total', 'saudiArabia', 'uae', 'uk', 'usa', 'otherGcc', 'eu']
    : dataset.id === 'exchange-rates' ? ['USD', 'EUR', 'GBP', 'CNY'] : [];
  for (const field of fields) {
    const latest = [...(data.monthly || [])].reverse().find(row => Number.isFinite(row[field]));
    add(field, field, latest?.date, latest?.evidence?.[field] || latest?.evidence || null, 'Monthly', latest?.status);
  }
  if (data.itHeadline) add('it-headline', 'IT & Telecom headline', data.itHeadline.latestMonth, data.itHeadline.evidence, 'Monthly', data.itHeadline.status);
  if (data.itMonthly) add('it-detail', 'Detailed IT components', data.itMonthly.detailLatestMonth || data.itMonthly.latestMonth, data.itMonthly.evidence, 'Monthly');
  if (data.sectorPeriod) add('fdi-sectors', 'FDI sector breakdown', fiscalPeriodEndDate(data.sectorPeriod), data.sectorEvidence || data.sourceEvidence?.['Netinflow.xls'], 'Monthly');
  if (data.countryPeriod) add('country-breakdown', 'Country breakdown', fiscalPeriodEndDate(data.countryPeriod), data.countryEvidence || data.sourceEvidence?.['Netinflow.xls'], 'Monthly');
  if (data.exportCountryPeriod) add('export-countries', 'Export country breakdown', fiscalPeriodEndDate(data.exportCountryPeriod), data.sourceEvidence?.['Export_Receipts_by_all_Countries.xls'], 'Monthly');
  if (data.importCountryPeriod) add('import-countries', 'Import country breakdown', fiscalPeriodEndDate(data.importCountryPeriod), data.sourceEvidence?.['Import-Payments-by-All-Countries.xlsx'], 'Monthly');
  for (const indicator of data.indicators || []) {
    if (!Array.isArray(indicator.values)) continue;
    for (const row of indicator.values) add(`${indicator.id}.${row.countryCode}`, `${indicator.label}: ${row.countryName}`, row.year ? String(row.year) : null, row.evidence || indicator.evidence, 'Annual', row.status);
  }
  return results;
}

export function getDatasetFreshness(dataset, data, { now = new Date(), checks = {}, validation = null } = {}) {
  const publication = data.publication || { policy: 'official-only', status: 'published', reason: null };
  const withheld = publication.status === 'withheld';
  const latestObservation = withheld ? null : dataset.latest?.(data) || data.observationDate || null;
  const observationDate = withheld ? null : dataset.latestDate?.(data) || latestObservation;
  const publicationDate = dataset.published?.(data) || explicitPublicationDate(data);
  const maxAgeDays = MAX_AGE_DAYS[dataset.cadence];
  const observationAge = observationAgeDays(observationDate, now);
  const stale = maxAgeDays != null && observationAge != null && observationAge > maxAgeDays;
  const undated = maxAgeDays != null && observationAge == null;
  const reviewRequired = data.reviewRequired === true;
  // A data file may downgrade its own trust tier at runtime — the FBR dataset
  // does exactly this whenever the latest month is only available through press
  // reporting of provisional figures.
  const sourceType = withheld ? 'unavailable' : data.sourceType || data.fytd?.sourceType || dataset.sourceType || 'unverified';
  const evidence = data.evidence || data.sourceEvidence || data.publicationEvidence || null;
  const hasReceipt = value => {
    if (!value || typeof value !== 'object') return false;
    if (value.artifactId && /^[a-f0-9]{64}$/.test(value.sha256 || value.evidenceHash || '') && officialDomain(value.sourceUrl)) return true;
    return Object.values(value).some(hasReceipt);
  };
  const authenticity = withheld ? 'unavailable'
    : hasReceipt(evidence) && ['official-primary', 'official-derived'].includes(sourceType) ? sourceType : 'unverified';
  const sourceKeys = new Set([dataset.sourceFile, ...(dataset.apiSeries || [])].filter(Boolean)
    .flatMap(key => key.includes('*') ? Object.keys(checks).filter(candidate => candidate.startsWith(key.split('*')[0])) : [key]));
  const gatherEvidence = value => {
    if (!value || typeof value !== 'object') return;
    if (value.artifactId && value.sourceKey) sourceKeys.add(value.sourceKey);
    for (const item of Object.values(value)) gatherEvidence(item);
  };
  gatherEvidence(data);
  const keys = [...sourceKeys];
  const checkDates = keys.filter(key => checks[key]?.status === 'success').map(key => checks[key].checkedAt).filter(Boolean).sort();
  const verifiedAt = checkDates.at(-1) || verificationDate(data);
  const sourceFailed = keys.some(key => checks[key]?.status === 'failed');
  const release = buildReleaseRow({ dataset, data, freshness: { latestObservation, dashboardUpdated: data.lastUpdated, publicationDate }, now });
  const eventCheckAge = observationAgeDays(verifiedAt?.slice(0, 10), now);
  const eventCheckOverdue = dataset.cadence === 'Event-driven' && (eventCheckAge == null || eventCheckAge > 14);
  const freshnessStatus = withheld ? 'withheld' : sourceFailed ? 'fetch-failed'
    : !latestObservation ? 'missing'
    : stale || (release.status === 'overdue' && release.schedule === 'announced') || data.newerOfficialReleaseAvailable === true ? 'overdue'
    : undated || reviewRequired || eventCheckOverdue ? 'needs-review'
    : 'latest-available';
  const checkedValidation = withheld ? { status: 'pending', checkedAt: null, checks: [] }
    : validation || { status: 'pending', checkedAt: null, checks: [] };
  return {
    id: dataset.id,
    label: dataset.label,
    file: dataset.file,
    source: dataset.source,
    sourceUrl: dataset.sourceUrl,
    sourceFile: dataset.sourceFile || null,
    sourceType,
    authenticity,
    publication,
    freshnessStatus,
    validation: checkedValidation,
    series: withheld ? [] : nestedSeries(dataset, data, now),
    evidence,
    expectedRelease: release.expectedRelease,
    releaseSchedule: release.schedule,
    releaseExpectation: release.status,
    sourceCheck: sourceFailed ? 'failed' : keys.length && keys.every(key => checks[key]?.status === 'success') ? 'success' : 'not-recorded',
    sourceLabel: data.sourceLabel || data.fytd?.sourceLabel || null,
    verifiedFrom: Array.isArray(data.verifiedFrom) ? data.verifiedFrom : null,
    apiSeries: dataset.apiSeries || null,
    parser: dataset.parser,
    cadence: dataset.cadence,
    expectedLag: dataset.expectedLag,
    critical: dataset.critical,
    latestObservation,
    observationDate,
    publicationDate,
    verificationDate: verifiedAt,
    freshnessDate: observationDate !== latestObservation ? observationDate : null,
    dataCoverage: data.dataCoverage || null,
    dashboardUpdated: data.lastUpdated || null,
    reviewReason: data.reviewReason || null,
    status: withheld ? 'withheld' : freshnessStatus === 'latest-available' ? 'fresh' : 'needs-review',
  };
}
