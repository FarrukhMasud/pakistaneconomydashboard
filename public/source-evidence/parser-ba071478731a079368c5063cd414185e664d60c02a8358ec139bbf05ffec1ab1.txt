// ─── Helpers ───

const workbookEvidence = new WeakMap();
const sheetEvidence = new WeakMap();

async function sourceReceipt(filename) {
  const { getSourceEvidence } = await import('./lib/source-evidence.mjs');
  const body = await readFile(resolve(RAW_DIR, filename));
  return { body, receipt: verifyReceipt(await getSourceEvidence(filename), body, filename) };
}

async function readExcel(filename) {
  const { body, receipt } = await sourceReceipt(filename);
  const wb = XLSX.read(body, { type: 'buffer' });
  workbookEvidence.set(wb, receipt);
  return wb;
}

function getSheet(wb, name) {
  const ws = wb.Sheets[name];
  if (!ws) throw new Error(`Sheet "${name}" not found. Available: ${wb.SheetNames.join(', ')}`);
  const rows = XLSX.utils.sheet_to_json(ws, { header: 1, defval: null, range: 0 });
  sheetEvidence.set(rows, { receipt: workbookEvidence.get(wb), sheet: name });
  return rows;
}

function excelDateToYYYYMM(serial) {
  sourceNumber(serial, { context: 'Excel date serial' });
  const d = XLSX.SSF.parse_date_code(serial);
  if (!d) throw new Error(`Invalid Excel date serial ${serial}`);
  calendarDate(`${d.y}-${String(d.m).padStart(2, '0')}-${String(d.d).padStart(2, '0')}`);
  return `${d.y}-${String(d.m).padStart(2, '0')}`;
}

function round2(n) {
  return n === null ? null : Math.round(sourceNumber(n) * 100) / 100;
}

function cellEvidence(rows, row, column, extra = {}) {
  const { receipt, sheet } = sheetEvidence.get(rows) || {};
  requireColumn(column, { sheet, row });
  return sourceEvidence(receipt, { sheet, row: row + 1, column: column + 1, cell: `${excelCell(column)}${row + 1}` }, { rawValue: rows[row]?.[column] ?? null, ...extra });
}

function cellNumber(rows, row, column, { required = true, divisor = 1 } = {}) {
  requireColumn(column, { row });
  const value = sourceNumber(rows[row]?.[column], { required, context: `${sheetEvidence.get(rows)?.sheet} ${excelCell(column)}${row + 1}` });
  return value === null ? null : value / divisor;
}

function requireUnit(rows, pattern) {
  const matches = rows.flatMap((row, r) => row.flatMap((value, c) => typeof value === 'string' && pattern.test(value) ? [{ row: r, column: c, label: value }] : []));
  if (!matches.length) throw new Error(`Missing printed source unit ${pattern}`);
  return matches[0];
}

function derivedEvidence(inputs, derivation) {
  if (!inputs.length || inputs.some(input => !input)) throw new Error(`Incomplete source evidence for derivation: ${derivation}`);
  return { derivation, inputs };
}

function bopSourceBlocks(rows, pattern, basis) {
  requireUnit(rows, /Million\s+US\s*\$/i);
  const headerIdx = requireRowIndex(rows, /^items$/i, { to: 20 });
  const row = requireRowIndex(rows, pattern);
  const width = Math.max(rows[headerIdx].length, rows[headerIdx + 1].length);
  const periods = forwardFill(rows[headerIdx], width);
  const blocks = [];
  for (let column = 1; column < width; column++) {
    const yearLabel = rows[headerIdx + 1][column];
    if (yearLabel === null || yearLabel === undefined || yearLabel === '') continue;
    const fy = parseFullFy(yearLabel);
    if (fy === null) throw new Error(`Invalid BOP fiscal year: ${yearLabel}`);
    const period = periods[column];
    const window = /[-\u2013]/.test(period) ? fiscalWindow(period, fy)
      : { start: fiscalMonth(period, fy), end: fiscalMonth(period, fy), fiscalYear: fy };
    blocks.push({
      ...window, column, period, basis, unit: 'US$ million',
      value: cellNumber(rows, row, column),
      status: publicationStatus(yearLabel), sourceStatus: yearLabel,
      evidence: cellEvidence(rows, row, column),
    });
  }
  return blocks;
}

export async function reconcileBopObservations(observations, fields, { loadWorkbook = readExcel } = {}) {
  const wb = await loadWorkbook('Balancepayment_BPM6.xls');
  const rows = getSheet(wb, 'BPM6_Summary');
  return reconcileBopRows(rows, observations, fields);
}

function reconcileBopRows(rows, observations, fields) {
  const sorted = uniqueObservations(observations);
  for (const row of sorted) calendarDate(row.date, { monthly: true });
  const checks = [];
  for (const [field, pattern, basis] of fields) {
    for (const block of bopSourceBlocks(rows, pattern, basis)) {
      const matches = sorted.filter(row => row.date >= block.start && row.date <= block.end);
      if (!matches.length || matches[0].date !== block.start || matches.at(-1).date !== block.end) {
        checks.push({ check: `Independent SBP BOP ${field}`, result: 'not-comparable', reason: 'different-coverage', rightPeriod: [block.start, block.end], rightEvidence: block.evidence });
        continue;
      }
      if (!consecutiveMonths(matches, 'date', { allowGaps: true })) {
        checks.push({ check: `Independent SBP BOP ${field}`, result: 'not-comparable', reason: 'incomplete-coverage', rightEvidence: block.evidence });
        continue;
      }
      const value = sumComplete(matches.map(row => row[field]));
      if (value === null || matches.some(row => !row.evidence?.[field])) {
        checks.push({ check: `Independent SBP BOP ${field}`, result: 'not-comparable', reason: value === null ? 'missing-observations' : 'missing-source-evidence', rightEvidence: block.evidence });
        continue;
      }
      checks.push(compareEquivalent(`Independent SBP BOP ${field}`, {
        start: block.start, end: block.end, basis, unit: block.unit,
        value,
        evidence: derivedEvidence(matches.map(row => row.evidence[field]), 'sum over exact printed BOP window'),
      }, block, { tolerance: matches.length * 0.5 + 0.5 }));
    }
  }
  return checks;
}

function directionalSentiment(trend, positiveWhenUp = true) {
  if (trend === 'stable' || trend === 'unavailable') return 'neutral';
  const positive = positiveWhenUp ? trend === 'up' : trend === 'down';
  return positive ? 'positive' : 'negative';
}

function targetBandSentiment(value, trend, min, max) {
  if (value >= min && value <= max) return 'positive';
  if (value > max) return trend === 'down' ? 'positive' : 'negative';
  return trend === 'up' ? 'positive' : 'negative';
}

/** Months in fiscal-year order, so July sorts before January. */
const FISCAL_MONTH_ORDER = ['jul', 'aug', 'sep', 'oct', 'nov', 'dec', 'jan', 'feb', 'mar', 'apr', 'may', 'jun'];

function fiscalPeriodEndIndex(period) {
  const match = String(period || '').match(/jul(?:y)?[-\s]+([a-z]+)/i);
  if (!match) return null;
  const end = match[1].slice(0, 3).toLowerCase();
  const index = FISCAL_MONTH_ORDER.indexOf(end);
  return index >= 0 ? index : null;
}

/**
 * SBP spells the same cumulative span differently per workbook ("Jul-May",
 * "July-June FY26", "July-June-FY26"). Render one consistent label.
 */
function formatFyPeriodLabel(period) {
  const raw = String(period || '').trim();
  const match = raw.match(/jul(?:y)?[-\s]+([a-z]+)/i);
  if (!match) return raw;
  const cap = (s) => `${s.charAt(0).toUpperCase()}${s.slice(1)}`;
  const span = `Jul-${cap(match[1].slice(0, 3).toLowerCase())}`;
  const fy = raw.match(/FY\s*-?\s*(\d{2,4})/i);
  return fy ? `${span} FY${fy[1]}` : span;
}

/**
 * SBP spells the same cumulative span differently per workbook ("Jul-May",
 * "July-June FY26"). Reduce it to a bare month span so the fiscal year is
 * carried once, by the label, and never printed twice in the UI.
 */
function normaliseFytdPeriod(period) {
  const raw = String(period || '').trim();
  // First month of a new FY is often published as a standalone "July 2026".
  if (/^jul(?:y)?[-\s]*\d{4}/i.test(raw) && !/jul(?:y)?[-\s]+[a-z]/i.test(raw)) {
    return 'Jul-Jul';
  }
  const match = raw.match(/jul(?:y)?[-\s]+([a-z]+)/i);
  if (!match) return raw;
  const end = match[1].slice(0, 3).toLowerCase();
  return `Jul-${end.charAt(0).toUpperCase()}${end.slice(1)}`;
}

function isLegacyExcel(buffer) {
  const signature = [0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1];
  return signature.every((byte, index) => buffer[index] === byte);
}

  const raw = await readFile(resolve(DATA_DIR, filename), 'utf-8');
  return JSON.parse(raw);
}

  );
}

async function fileExists(filepath) {
  try { await access(filepath); return true; } catch (error) {
    if (error.code === 'ENOENT') return false;
    throw error;
  }
}

function getFlag(country) {
  const c = country.toLowerCase().trim();
  // Direct match
  if (FLAG_MAP[c]) return FLAG_MAP[c];
  // Normalize dots and spaces: "U. S. A." → "usa", "U. K." → "uk", "U. A. E. Dubai" → "uae dubai"
  const normalized = c.replace(/\.\s*/g, '').replace(/\s+/g, ' ');
  if (FLAG_MAP[normalized]) return FLAG_MAP[normalized];
  // Partial matches for common SBP abbreviations
  if (normalized.startsWith('u s a') || normalized.startsWith('usa')) return '🇺🇸';
  if (normalized.startsWith('u k') || normalized === 'uk') return '🇬🇧';
  if (normalized.startsWith('uae') || normalized.startsWith('u a e')) return '🇦🇪';
  if (normalized.includes('korea')) return '🇰🇷';
  if (normalized.includes('hong kong')) return '🇭🇰';
  // Try matching a substring of the country name against FLAG_MAP keys
  for (const [key, flag] of Object.entries(FLAG_MAP)) {
    if (normalized.includes(key) || key.includes(normalized)) return flag;
  }
  return '🏳️';
}

async function updateTrade({ loadWorkbook = readExcel, readData = readJson, writeData = writeJson } = {}) {
  console.log('\n📦 Parsing Trade Data (exp_import_BOP.xls)...');

  const wb = await loadWorkbook('exp_import_BOP.xls');
  const sheetName = wb.SheetNames.find(name =>
    ['Exp.Imp.(BOP)Arch', 'Exp.Imp.(BOP)'].includes(name.trim()),
  );
  if (!sheetName) {
    throw new Error(`Trade sheet not found. Available: ${wb.SheetNames.join(', ')}`);
  }
  const rows = getSheet(wb, sheetName);
  requireUnit(rows, /Million\s+US\s*\$/i);
  const headerIndex = rows.findIndex(row => row.some(cell => /^Period$/i.test(String(cell).trim())));
  if (headerIndex < 0) throw new Error('Trade source has no Period header');
  const header = rows[headerIndex];
  const dateCol = requireColumn(header.findIndex(cell => /^Period$/i.test(String(cell).trim())));
  const exportsCol = requireColumn(header.findIndex(cell => /^Exports\s*\(BOP\)$/i.test(String(cell).trim())));
  const importsCol = requireColumn(header.findIndex(cell => /^Imports\s*\(BOP\)$/i.test(String(cell).trim())));
  const balanceCol = requireColumn(header.findIndex(cell => /^Balance of Trade$/i.test(String(cell).trim())));
  if (!/^Value$/i.test(String(rows[headerIndex + 1]?.[exportsCol])) || !/^Value$/i.test(String(rows[headerIndex + 1]?.[importsCol]))) {
    throw new Error('Trade source lacks monthly Value subheaders');
  }
  const statusCol = dateCol + 1;

  // The current SBP archive adds a leading blank column and a revision-status
  // column. The legacy workbook remains supported as a fallback source.
  const monthly = [];

  const hasFormulaHeader = rows[headerIndex + 2]?.[exportsCol] === 'a'
    && rows[headerIndex + 2]?.[importsCol] === 'c' && rows[headerIndex + 2]?.[balanceCol] === 'a-c';
  for (let i = headerIndex + (hasFormulaHeader ? 3 : 2); i < rows.length; i++) {
    const row = rows[i];
    if (!row) continue;
    if (typeof row[dateCol] !== 'number') {
      if ([exportsCol, importsCol, balanceCol].some(col => row[col] !== null && row[col] !== undefined && row[col] !== '')) {
        fiscalYearRange(row[dateCol], { context: `Trade period ${excelCell(dateCol)}${i + 1}` });
      }
      continue;
    }

    const date = excelDateToYYYYMM(row[dateCol]);
    if (parseInt(date.substring(0, 4)) < 2021) continue;
    const exports_ = cellNumber(rows, i, exportsCol);
    const imports_ = cellNumber(rows, i, importsCol);
    const balance = cellNumber(rows, i, balanceCol);
    assertReconciled(`Trade balance ${date}`, balance, [exports_, -imports_], { tolerance: 0.02 });

    monthly.push({
      date,
      imports: Math.round(imports_),
      exports: Math.round(exports_),
      balance: Math.round(balance),
      status: publicationStatus(row[statusCol]),
      sourceStatus: row[statusCol] ?? null,
      unit: 'US$ million',
      evidence: {
        exports: cellEvidence(rows, i, exportsCol), imports: cellEvidence(rows, i, importsCol),
        balance: cellEvidence(rows, i, balanceCol),
      },
    });
  }

  monthly.sort((a, b) => a.date.localeCompare(b.date));
  if (!monthly.length) throw new Error('Trade source contains no monthly observations');
  if (new Set(monthly.map(row => row.date)).size !== monthly.length) {
    throw new Error('Trade source contains duplicate monthly observations');
  }
  const reconciliation = await reconcileBopObservations(monthly, [
    ['exports', /^Exports of Goods FOB$/i, 'BOP-goods-exports'],
    ['imports', /^Imports of Goods FOB$/i, 'BOP-goods-imports'],
  ], { loadWorkbook });

  const existing = await readData('trade.json');
  const history = retainHistory(existing.monthly || [], monthly);
  const firstDate = history[0]?.date;
  const lastDate = history.at(-1)?.date;
  await writeData('trade.json', {
    ...existing,
    monthly: history,
    reconciliation,
    topExportCountries: existing.topExportCountries || [],
    topImportCountries: existing.topImportCountries || [],
    exportCountryPeriod: existing.exportCountryPeriod || null,
    importCountryPeriod: existing.importCountryPeriod || null,
    dataSource: 'SBP',
    lastUpdated: new Date().toISOString().split('T')[0],
    dataCoverage: `${firstDate} – ${lastDate}`,
  });

  console.log(`  📊 ${monthly.length} months (${firstDate} → ${lastDate})`);
  return monthly.length;
}

// ═══════════════════════════════════════════════════
// 2. FDI (Netinflow sector/country, NetinflowSummary annual)
// ═══════════════════════════════════════════════════

async function updateFdi({ loadWorkbook = readExcel, readData = readJson, writeData = writeJson } = {}) {
  console.log('\n💰 Parsing FDI Data...');

  // --- By Sector (Netinflow.xls / Sector sheet) ---
  // SBP publishes two sector cuts: Foreign_Dir.xls carries broad ISIC letter
  // sections, Netinflow.xls carries its own detailed sector list. The detailed
  // sheet is refreshed earlier in the month and its rows sum exactly to the
  // published total, so it is the source of record here. Row 5 = period block,
  // row 6 = Inflow/Outflow/Net FDI sub-headers, rows 7+ = data.
  console.log('  📋 FDI by sector (Netinflow.xls)...');
  const wbS = await loadWorkbook('Netinflow.xls');
  const sRows = getSheet(wbS, 'Sector');
  requireUnit(sRows, /Million\s+US\s*\$/i);

  // Columns are resolved from the printed fiscal year, never by position: if
  // SBP moves these columns we must fail loudly instead of silently publishing
  // a different period's numbers.
  const sectorCols = resolveFdiCountryColumns(sRows[5] || [], sRows[6] || [], sRows[6] || [], {
    file: 'Netinflow.xls',
    sheet: 'Sector',
  });
  const sCurrentInCol = sectorCols.current.inflow;
  const sCurrentOutCol = sectorCols.current.outflow;
  const sCurrentNetCol = sectorCols.current.net;
  const sPriorInCol = sectorCols.prior.inflow;
  const sPriorOutCol = sectorCols.prior.outflow;
  const sPriorNetCol = sectorCols.prior.net;
  const sCurrentPeriod = sectorCols.current.period;
  const sPriorPeriod = sectorCols.prior.period;
  const sectorStatus = {
    current: publicationStatus(forwardFill(sRows[5])[sCurrentNetCol]),
    prior: publicationStatus(forwardFill(sRows[5])[sPriorNetCol]),
  };
  const sectorWindow = fiscalWindow(sCurrentPeriod, sectorCols.current.fiscalYear);
  const sectorPriorWindow = fiscalWindow(sPriorPeriod, sectorCols.prior.fiscalYear);
  if (sectorWindow.end.slice(5) !== sectorPriorWindow.end.slice(5)) throw new Error('FDI sector comparison uses different FY windows');
  console.log(`  Sector columns: current ${sCurrentPeriod} @${sCurrentNetCol}, prior ${sPriorPeriod} @${sPriorNetCol}`);

  // The sheet nests breakdowns under their parent ("Power" is followed by
  // "I) Thermal", "II) Hydel"). Only top-level rows are summed, otherwise the
  // sector table would double-count and stop reconciling with the total.
  const SECTOR_SUBROW = /^(?:[IVX]+\)|\d+\)|of which)/i;
  const allSectors = [];
  for (let i = 7; i < sRows.length; i++) {
    const row = sRows[i];
    if (!row) continue;
    if (/^total$/i.test(String(row[1]).trim())) break;
    const sector = (row[2] || '').toString().trim();
    if (!sector) continue;
    if (/^total/i.test(sector)) break;
    if (SECTOR_SUBROW.test(sector)) continue;

    const netFdi = cellNumber(sRows, i, sCurrentNetCol);

    const entry = {
      sector: shortenSector(sector),
      amount: round2(netFdi),
      inflow: round2(cellNumber(sRows, i, sCurrentInCol)),
      outflow: round2(cellNumber(sRows, i, sCurrentOutCol)),
      status: sectorStatus.current, priorStatus: sectorStatus.prior,
      evidence: {
        amount: cellEvidence(sRows, i, sCurrentNetCol),
        inflow: cellEvidence(sRows, i, sCurrentInCol), outflow: cellEvidence(sRows, i, sCurrentOutCol),
        priorAmount: cellEvidence(sRows, i, sPriorNetCol),
        priorInflow: cellEvidence(sRows, i, sPriorInCol), priorOutflow: cellEvidence(sRows, i, sPriorOutCol),
      },
    };
    // Prior year comparison
    const priorNet = cellNumber(sRows, i, sPriorNetCol);
    {
      entry.priorAmount = round2(priorNet);
      entry.priorInflow = round2(cellNumber(sRows, i, sPriorInCol));
      entry.priorOutflow = round2(cellNumber(sRows, i, sPriorOutCol));
    }
    assertReconciled(`FDI sector ${sector} current net`, netFdi, [cellNumber(sRows, i, sCurrentInCol), -cellNumber(sRows, i, sCurrentOutCol)]);
    assertReconciled(`FDI sector ${sector} prior net`, priorNet, [cellNumber(sRows, i, sPriorInCol), -cellNumber(sRows, i, sPriorOutCol)]);
    allSectors.push(entry);
  }

  // Stable sector universe: rank by max absolute net across both years
  allSectors.sort((a, b) => {
    const magA = Math.max(Math.abs(a.amount), Math.abs(a.priorAmount));
    const magB = Math.max(Math.abs(b.amount), Math.abs(b.priorAmount));
    return magB - magA;
  });
  // The sheet already publishes its own "Others" line; fold it into the
  // remainder bucket so the chart cannot show two rows with the same name.
  const sectorTotalRow = requireRowIndex(sRows, /^total$/i, { labelCol: 1, from: 7 });
  const sectorReconciliation = [
    assertReconciled('FDI sector current components', cellNumber(sRows, sectorTotalRow, sCurrentNetCol), allSectors.map(entry => entry.amount), { tolerance: allSectors.length * 0.005 + 0.01 }),
    assertReconciled('FDI sector prior components', cellNumber(sRows, sectorTotalRow, sPriorNetCol), allSectors.map(entry => entry.priorAmount), { tolerance: allSectors.length * 0.005 + 0.01 }),
  ];
  const rawOtherSectorIdx = allSectors.findIndex((s) => /^others$/i.test(s.sector));
  const rawOtherSector = rawOtherSectorIdx >= 0 ? allSectors.splice(rawOtherSectorIdx, 1)[0] : null;
  const topSectors = allSectors.slice(0, 10);
  const otherSectors = [...allSectors.slice(10), ...(rawOtherSector ? [rawOtherSector] : [])];
  const otherSectorAmt = otherSectors.reduce((s, x) => s + x.amount, 0);
  const otherSectorPrior = otherSectors.reduce((s, x) => s + x.priorAmount, 0);
  if (Math.abs(otherSectorAmt) > 0.01 || Math.abs(otherSectorPrior) > 0.01) {
    topSectors.push({
      sector: 'Others',
      amount: round2(otherSectorAmt),
      inflow: null, outflow: null,
      priorAmount: round2(otherSectorPrior),
      priorInflow: null, priorOutflow: null,
      status: sectorStatus.current, priorStatus: sectorStatus.prior,
      evidence: {
        amount: derivedEvidence(otherSectors.map(entry => entry.evidence.amount), 'sum of remainder sectors'),
        priorAmount: derivedEvidence(otherSectors.map(entry => entry.evidence.priorAmount), 'sum of remainder sectors'),
      },
    });
  }

  // --- By Country (Netinflow.xls / Country sheet) ---
  // Row 3: period headers, Row 4: sub-headers, Row 6+: data
  console.log('  📋 FDI by country (Netinflow.xls)...');
  const wbC = await loadWorkbook('Netinflow.xls');
  const cRows = getSheet(wbC, 'Country');
  requireUnit(cRows, /Million\s+US\s*\$/i);

  // Resolve column indices from header rows. Current vs prior is decided by the
  // fiscal year printed in the header, not by column order or the (P) marker.
  // Row 3 = period block, row 4 = instrument (FDI / FPI / Total),
  // row 5 = FDI sub-column (Inflow / Outflow / Net).
  const countryCols = resolveFdiCountryColumns(cRows[3] || [], cRows[4] || [], cRows[5] || [], {
    file: 'Netinflow.xls',
    sheet: 'Country',
  });
  const cCurrentNetCol = countryCols.current.net;
  const cPriorNetCol = countryCols.prior.net;
  const cCurrentPeriod = countryCols.current.period;
  const cPriorPeriod = countryCols.prior.period;
  const countryWindow = fiscalWindow(cCurrentPeriod, countryCols.current.fiscalYear);
  const countryPriorWindow = fiscalWindow(cPriorPeriod, countryCols.prior.fiscalYear);
  if (countryWindow.end.slice(5) !== countryPriorWindow.end.slice(5)) throw new Error('FDI country comparison uses different FY windows');
  const countryStatus = {
    current: publicationStatus(forwardFill(cRows[3])[cCurrentNetCol]),
    prior: publicationStatus(forwardFill(cRows[3])[cPriorNetCol]),
  };
  console.log(`  Country columns: current ${cCurrentPeriod} @${cCurrentNetCol}, prior ${cPriorPeriod} @${cPriorNetCol}`);

  const allCountries = [];
  const countryNames = new Set();
  for (let i = 6; i < cRows.length; i++) {
    const row = cRows[i];
    if (!row || typeof row[0] !== 'number') break;

    const country = (row[1] || '').toString().trim();
    const netFdi = cellNumber(cRows, i, cCurrentNetCol);
    if (!country) throw new Error(`FDI country missing label at row ${i + 1}`);
    if (countryNames.has(country)) throw new Error(`Duplicate FDI country: ${country}`);
    countryNames.add(country);

    const entry = {
      country, amount: round2(netFdi), flag: getFlag(country),
      status: countryStatus.current, priorStatus: countryStatus.prior,
      evidence: { amount: cellEvidence(cRows, i, cCurrentNetCol), priorAmount: cellEvidence(cRows, i, cPriorNetCol) },
    };
    const priorNet = cellNumber(cRows, i, cPriorNetCol);
    entry.priorAmount = round2(priorNet);
    for (const group of [countryCols.current, countryCols.prior]) {
      assertReconciled(`FDI country ${country} FY${group.fiscalYear} net`, cellNumber(cRows, i, group.net),
        [cellNumber(cRows, i, group.inflow), -cellNumber(cRows, i, group.outflow)]);
    }
    allCountries.push(entry);
  }

  allCountries.sort((a, b) => Math.abs(b.amount) - Math.abs(a.amount));

  // The country sheet's own "Total" line is the same measure the summary
  // workbook publishes (net FDI for the fiscal year to date), but SBP refreshes
  // Netinflow.xls earlier in the month than NetinflowSummary.xls. Capture it so
  // the headline can prefer whichever official file covers the later period.
  let countryFytd = null;
  const reconciliation = [...sectorReconciliation];
  const totalRowIdx = requireRowIndex(cRows, /^total$/i, { labelCol: 0, from: 6 });
  if (totalRowIdx >= 0) {
    const readCol = (col) => round2(cellNumber(cRows, totalRowIdx, col));
    const currentNet = readCol(cCurrentNetCol);
    const priorNet = readCol(cPriorNetCol);
    if (currentNet != null) {
      countryFytd = {
        period: normaliseFytdPeriod(cCurrentPeriod),
        current: {
          label: `FY${countryCols.current.fiscalYear}`,
          net_fdi: currentNet,
          inflow: readCol(countryCols.current.inflow),
          outflow: readCol(countryCols.current.outflow),
          status: countryStatus.current,
          evidence: {
            net_fdi: cellEvidence(cRows, totalRowIdx, cCurrentNetCol),
            inflow: cellEvidence(cRows, totalRowIdx, countryCols.current.inflow),
            outflow: cellEvidence(cRows, totalRowIdx, countryCols.current.outflow),
          },
        },
        prior: priorNet != null ? {
          label: `FY${countryCols.prior.fiscalYear}`,
          net_fdi: priorNet,
          inflow: readCol(countryCols.prior.inflow),
          outflow: readCol(countryCols.prior.outflow),
          status: countryStatus.prior,
          evidence: {
            net_fdi: cellEvidence(cRows, totalRowIdx, cPriorNetCol),
            inflow: cellEvidence(cRows, totalRowIdx, countryCols.prior.inflow),
            outflow: cellEvidence(cRows, totalRowIdx, countryCols.prior.outflow),
          },
        } : null,
        sourceFile: 'Netinflow.xls',
        sourceSheet: 'Country',
        sourceLocation: '"Total" row, net FDI column of the current fiscal-year-to-date block',
      };
      console.log(`  Country totals: ${cCurrentPeriod} net $${currentNet}M (prior ${cPriorPeriod} $${priorNet}M)`);
    }
    reconciliation.push(
      assertReconciled('FDI country current components', countryFytd.current.net_fdi, allCountries.map(entry => entry.amount), { tolerance: allCountries.length * 0.005 + 0.01 }),
      assertReconciled('FDI country prior components', countryFytd.prior.net_fdi, allCountries.map(entry => entry.priorAmount), { tolerance: allCountries.length * 0.005 + 0.01 }),
    );
    for (const [countryGroup, sectorGroup, window, counterpart] of [
      [countryCols.current, sectorCols.current, countryWindow, sectorWindow],
      [countryCols.prior, sectorCols.prior, countryPriorWindow, sectorPriorWindow],
    ]) {
      reconciliation.push(compareEquivalent('FDI country total vs independent sector total', {
        ...window, basis: 'net-FDI-in-Pakistan', unit: 'US$ million', value: cellNumber(cRows, totalRowIdx, countryGroup.net), evidence: cellEvidence(cRows, totalRowIdx, countryGroup.net),
      }, {
        ...counterpart, basis: 'net-FDI-in-Pakistan', unit: 'US$ million', value: cellNumber(sRows, sectorTotalRow, sectorGroup.net), evidence: cellEvidence(sRows, sectorTotalRow, sectorGroup.net),
      }, { tolerance: 0.02 }));
    }
  }

  // Separate raw "Others" from named countries, then aggregate remainder into "Others"
  const rawOthersIdx = allCountries.findIndex(c => c.country.toLowerCase() === 'others');
  const rawOthers = rawOthersIdx >= 0 ? allCountries.splice(rawOthersIdx, 1)[0] : null;
  const topCountries = allCountries.slice(0, 10);
  const remainderCountries = [...allCountries.slice(10), ...(rawOthers ? [rawOthers] : [])];
  const remainderAmt = remainderCountries.reduce((s, x) => s + x.amount, 0);
  const remainderPrior = remainderCountries.reduce((s, x) => s + x.priorAmount, 0);
  if (Math.abs(remainderAmt) > 0.01 || Math.abs(remainderPrior) > 0.01) {
    topCountries.push({
      country: 'Others', amount: round2(remainderAmt), flag: '🌍',
      priorAmount: round2(remainderPrior),
      status: countryStatus.current, priorStatus: countryStatus.prior,
      evidence: {
        amount: derivedEvidence(remainderCountries.map(entry => entry.evidence.amount), 'sum of remainder countries'),
        priorAmount: derivedEvidence(remainderCountries.map(entry => entry.evidence.priorAmount), 'sum of remainder countries'),
      },
    });
  }

  // --- Annual FDI (NetinflowSummary.xls / Summary sheet) ---
  // Rows are located by their own labels, not by fixed indices, so an inserted
  // row upstream can never silently shift which line we publish.
  console.log('  📋 Annual FDI (NetinflowSummary.xls)...');
  const wbA = await loadWorkbook('NetinflowSummary.xls');
  const aRows = getSheet(wbA, 'Summary');
  requireUnit(aRows, /Million\s+US\s*\$/i);

  const summaryCtx = { file: 'NetinflowSummary.xls', sheet: 'Summary' };
  const fdiRowIdx = requireRowIndex(aRows, /^Direct\s+Investment$/i, { labelCol: 1, ...summaryCtx });
  const inflowRowIdx = requireRowIndex(aRows, /^Inflow$/i, { labelCol: 3, from: fdiRowIdx, to: fdiRowIdx + 4 });
  const outflowRowIdx = requireRowIndex(aRows, /^Outflow$/i, { labelCol: 3, from: fdiRowIdx, to: fdiRowIdx + 4 });
  // The FY header block sits a few rows above the first data line.
  const headerRowIdx = requireRowIndex(aRows, /^FY\s*\d{2}/i, { labelCol: 4, to: fdiRowIdx }) ;
  const headerRow = aRows[headerRowIdx];
  const fdiRow = aRows[fdiRowIdx];     // Direct Investment (net)
  const inflowRow = aRows[inflowRowIdx];
  const outflowRow = aRows[outflowRowIdx];
  const subHeaderRow = aRows[headerRowIdx + 1] || [];
  console.log(`  Summary rows: header@${headerRowIdx}, net@${fdiRowIdx}, inflow@${inflowRowIdx}, outflow@${outflowRowIdx}`);

  const annual = [];
  let fytdComparison = null;
  let monthlyComparison = null;

  // Find the latest cumulative FYTD column (for example "Jul-Apr").
  // SBP places current FYTD first, with the prior-year comparison in the next column.
  let fytdCurrentCol = -1;
  let fytdPriorCol = -1;
  let fytdCurrentFy = null;
  let fytdPriorFy = null;
  let fytdPeriod = '';
  const toFullFy = parseFullFy;

  for (let col = 0; col < headerRow.length; col++) {
    const hdr = (headerRow[col] || '').toString().trim();

    // Full fiscal year columns
    const m = hdr.match(/^FY(\d{4}|\d{2})(?!\d)(?:\s*\([PRF]\))?\s*$/i);
    if (m) {
      const fy = parseFullFy(hdr);
      if (fy === null) throw new Error(`Invalid FDI annual fiscal year ${hdr}`);
      if (fy < 2017) continue;
      const val = cellNumber(aRows, fdiRowIdx, col);
      const status = publicationStatus(hdr);
      const entry = { year: `FY${fy}`, net_fdi: Math.round(val) };
      entry.inflow = Math.round(cellNumber(aRows, inflowRowIdx, col));
      entry.outflow = Math.round(cellNumber(aRows, outflowRowIdx, col));
      entry.status = status;
      entry.evidence = {
        net_fdi: cellEvidence(aRows, fdiRowIdx, col), inflow: cellEvidence(aRows, inflowRowIdx, col), outflow: cellEvidence(aRows, outflowRowIdx, col),
      };
      reconciliation.push(assertReconciled(`FDI annual FY${fy} net`, val, [cellNumber(aRows, inflowRowIdx, col), -cellNumber(aRows, outflowRowIdx, col)]));
      annual.push(entry);
    }

    if (
      fytdCurrentCol === -1 &&
      /^jul[-\s]/i.test(hdr) &&
      fiscalWindow(hdr, parseFullFy(subHeaderRow[col])).end
    ) {
      const firstMatch = (subHeaderRow[col] || '').toString().trim().match(/^FY(\d{2,4})/i);
      const secondMatch = (subHeaderRow[col + 1] || '').toString().trim().match(/^FY(\d{2,4})/i);

      if (firstMatch && secondMatch) {
        const firstFy = toFullFy(firstMatch[1]);
        const secondFy = toFullFy(secondMatch[1]);
        const currentIsFirst = firstFy > secondFy;
        fytdCurrentCol = currentIsFirst ? col : col + 1;
        fytdPriorCol = currentIsFirst ? col + 1 : col;
        fytdCurrentFy = Math.max(firstFy, secondFy);
        fytdPriorFy = Math.min(firstFy, secondFy);
      } else {
        throw new Error('FDI summary cumulative columns lack explicit fiscal years');
      }
      fytdPeriod = normaliseFytdPeriod(hdr.replace(/\s*\([RP]\)\s*/i, '').trim());
    }

    if (!monthlyComparison && /^(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)$/i.test(hdr)) {
      const priorHdr = (subHeaderRow[col] || '').toString().trim();
      const currentHdr = (subHeaderRow[col + 1] || '').toString().trim();
      const priorMatch = priorHdr.match(/^FY(\d{2,4})/i);
      const currentMatch = currentHdr.match(/^FY(\d{2,4})/i);

      if (!priorMatch || !currentMatch) throw new Error(`FDI monthly ${hdr} has no fiscal-year pair`);
      {
        const firstFy = toFullFy(priorMatch[1]);
        const secondFy = toFullFy(currentMatch[1]);
        const currentCol = firstFy > secondFy ? col : col + 1;
        const priorCol = currentCol === col ? col + 1 : col;
        const currentFy = Math.max(firstFy, secondFy);
        const priorFy = Math.min(firstFy, secondFy);
        if (currentFy !== priorFy + 1) throw new Error('FDI monthly comparison fiscal years are not consecutive');
        const block = (index, fy) => {
          const entry = {
            label: `FY${fy}`, net_fdi: round2(cellNumber(aRows, fdiRowIdx, index)),
            inflow: round2(cellNumber(aRows, inflowRowIdx, index)),
            outflow: round2(cellNumber(aRows, outflowRowIdx, index)),
            status: publicationStatus(subHeaderRow[index]),
            evidence: { net_fdi: cellEvidence(aRows, fdiRowIdx, index), inflow: cellEvidence(aRows, inflowRowIdx, index), outflow: cellEvidence(aRows, outflowRowIdx, index) },
          };
          reconciliation.push(assertReconciled(`FDI monthly ${hdr} FY${fy} net`, entry.net_fdi, [entry.inflow, -entry.outflow]));
          return entry;
        };

        monthlyComparison = {
          month: hdr,
          current: block(currentCol, currentFy), prior: block(priorCol, priorFy),
        };
      }
    }
  }

  // Extract FYTD comparison (e.g. Jul-Apr FY26 vs Jul-Apr FY25)
  if (fytdCurrentCol >= 0) {
    if (fytdCurrentFy !== fytdPriorFy + 1) throw new Error('FDI summary comparison fiscal years are not consecutive');

    const currentFyLabel = `FY${fytdCurrentFy}`;
    const priorFyLabel = `FY${fytdPriorFy}`;
    fytdComparison = {
      period: fytdPeriod,
      current: { label: currentFyLabel },
      prior: { label: priorFyLabel },
      sourceFile: 'NetinflowSummary.xls',
      sourceSheet: 'Summary',
      sourceLocation: '"Direct Investment" row, current fiscal-year-to-date column',
    };
    for (const [entry, col] of [[fytdComparison.current, fytdCurrentCol], [fytdComparison.prior, fytdPriorCol]]) {
      entry.net_fdi = round2(cellNumber(aRows, fdiRowIdx, col));
      entry.inflow = round2(cellNumber(aRows, inflowRowIdx, col));
      entry.outflow = round2(cellNumber(aRows, outflowRowIdx, col));
      entry.status = publicationStatus(subHeaderRow[col]);
      entry.evidence = { net_fdi: cellEvidence(aRows, fdiRowIdx, col), inflow: cellEvidence(aRows, inflowRowIdx, col), outflow: cellEvidence(aRows, outflowRowIdx, col) };
      reconciliation.push(assertReconciled(`FDI summary ${fytdPeriod} ${entry.label} net`, entry.net_fdi, [entry.inflow, -entry.outflow]));
      const countryEntry = entry.label === countryFytd?.current.label ? countryFytd.current
        : entry.label === countryFytd?.prior.label ? countryFytd.prior : null;
      if (countryEntry) reconciliation.push(compareEquivalent('FDI summary vs country total', {
        ...fiscalWindow(fytdPeriod, entry.label), basis: 'net-FDI-in-Pakistan', unit: 'US$ million', value: entry.net_fdi, evidence: entry.evidence.net_fdi,
      }, {
        ...fiscalWindow(countryFytd.period, countryEntry.label), basis: 'net-FDI-in-Pakistan', unit: 'US$ million', value: countryEntry.net_fdi, evidence: countryEntry.evidence.net_fdi,
      }, { tolerance: 0.02 }));
    }
  }

  // SBP publishes the same fiscal-year-to-date net FDI figure in two workbooks
  // on different schedules. Whichever covers the later month is the current
  // official figure, so the headline follows the period, never a fixed file.
  if (countryFytd) {
    const summaryEnd = fiscalPeriodEndIndex(fytdComparison?.period);
    const countryEnd = fiscalPeriodEndIndex(countryFytd.period);
    const fyNum = (label) => {
      const match = String(label || '').match(/(\d{4}|\d{2})$/);
      if (!match) return null;
      const n = parseInt(match[1], 10);
      return n < 100 ? 2000 + n : n;
    };
    const summaryFy = fyNum(fytdComparison?.current?.label);
    const countryFy = fyNum(countryFytd.current.label);
    const laterFy = countryFy != null && (summaryFy == null || countryFy > summaryFy);
    const sameFy = fytdComparison?.current?.label === countryFytd.current.label;
    if (
      !fytdComparison ||
      laterFy ||
      (sameFy && countryEnd != null && (summaryEnd == null || countryEnd > summaryEnd))
    ) {
      if (fytdComparison) {
        console.log(
          `  ↪ Headline FDI switched to ${countryFytd.sourceFile} (${countryFytd.period}) — ` +
          `${fytdComparison.sourceFile} still publishes ${fytdComparison.period}`,
        );
      }
      fytdComparison = countryFytd;
    }
  }

  // Once the fiscal year has closed, its Jul–Jun cumulative *is* the annual
  // figure. Publish it as soon as an official file carries the complete year
  // rather than waiting for the annual summary column to appear.
  if (fytdComparison && fiscalPeriodEndIndex(fytdComparison.period) === 11) {
    const year = fytdComparison.current.label;
    const entry = {
      year,
      net_fdi: Math.round(fytdComparison.current.net_fdi),
      status: fytdComparison.current.status,
      evidence: fytdComparison.current.evidence,
    };
    if (fytdComparison.current.inflow != null) entry.inflow = Math.round(fytdComparison.current.inflow);
    if (fytdComparison.current.outflow != null) entry.outflow = Math.round(fytdComparison.current.outflow);
    const existingIdx = annual.findIndex((a) => a.year === year);
    if (existingIdx >= 0) annual[existingIdx] = { ...annual[existingIdx], ...entry };
    else annual.push(entry);
    annual.sort((a, b) => parseInt(a.year.replace('FY', '')) - parseInt(b.year.replace('FY', '')));
    console.log(`  📊 Completed fiscal year ${year} published as annual: $${entry.net_fdi}M (${entry.status})`);
  }

  const existing = await readData('fdi.json');
  // NetinflowSummary lags the July FY rollover; keep completed years already
  // published from a closed Jul-Jun Netinflow.xls so they do not disappear.
  for (const entry of existing.annual || []) {
    if (!entry?.year || annual.some((a) => a.year === entry.year)) continue;
    annual.push(entry);
  }
  annual.sort((a, b) => parseInt(a.year.replace('FY', '')) - parseInt(b.year.replace('FY', '')));
  const existingFytd = existing.fytdComparison;
  if (
    fytdComparison &&
    existingFytd?.current?.label === fytdComparison.current.label
  ) {
    const existingEnd = fiscalPeriodEndIndex(existingFytd.period);
    const incomingEnd = fiscalPeriodEndIndex(fytdComparison.period);
    if (existingEnd != null && incomingEnd != null && incomingEnd < existingEnd) {
      throw new Error(
        `Refusing FDI period regression from ${existingFytd.period} to ${fytdComparison.period} for ${fytdComparison.current.label}`,
      );
    }
  }

  const result = {
    ...existing,
    by_sector: topSectors,
    by_country: topCountries,
    annual,
    reconciliation,
    sectorPeriod: formatFyPeriodLabel(sCurrentPeriod),
    sectorPriorPeriod: formatFyPeriodLabel(sPriorPeriod),
    countryPeriod: formatFyPeriodLabel(cCurrentPeriod),
    countryPriorPeriod: formatFyPeriodLabel(cPriorPeriod),
    source: 'State Bank of Pakistan',
    dataSource: 'SBP',
    lastUpdated: new Date().toISOString().split('T')[0],
  };
  if (fytdComparison) result.fytdComparison = fytdComparison;
  if (monthlyComparison) result.monthlyComparison = monthlyComparison;

  const coverageParts = [
    existing.monthly?.at(-1)?.date ? `monthly through ${existing.monthly.at(-1).date}` : null,
    `sector ${formatFyPeriodLabel(sCurrentPeriod)}`,
    `country ${formatFyPeriodLabel(cCurrentPeriod)}`,
    fytdComparison
      ? `headline ${fytdComparison.current.label} ${fytdComparison.period} (${fytdComparison.sourceFile})`
      : `annual ${annual[0]?.year} – ${annual.at(-1)?.year}`,
  ].filter(Boolean);
  result.dataCoverage = coverageParts.join('; ');

  await writeData('fdi.json', result);

  console.log(`  📊 ${topSectors.length} sectors, ${topCountries.length} countries, ${annual.length} fiscal years`);
  if (fytdComparison) console.log(`  📊 FYTD: ${fytdComparison.current.label} ${fytdComparison.period}: $${fytdComparison.current.net_fdi}M vs ${fytdComparison.prior?.label}: $${fytdComparison.prior?.net_fdi}M`);
  if (monthlyComparison) console.log(`  📊 Monthly: ${monthlyComparison.month} ${monthlyComparison.current.label}: $${monthlyComparison.current.net_fdi}M vs ${monthlyComparison.prior.label}: $${monthlyComparison.prior.net_fdi}M`);
  return { sectors: topSectors.length, countries: topCountries.length, years: annual.length };
}

// ═══════════════════════════════════════════════════
// 3. GDP / FISCAL (GDP_table.xlsx)
// ═══════════════════════════════════════════════════

async function updateGdpFiscal({ loadWorkbook = readExcel, readData = readJson, writeData = writeJson } = {}) {
  console.log('\n📈 Parsing GDP Data (GDP_table.xlsx)...');

  const wb = await loadWorkbook('GDP_table.xlsx');
  const rows = getSheet(wb, 'Annual');

  // The year header row and the growth row are located by label so an inserted
  // row or column upstream cannot silently shift which series we publish.
  const growthRowIdx = requireRowIndex(rows, /GDP\s+Growth\s+Rate/i, {
    labelCol: 2,
    to: 40,
    file: 'GDP_table.xlsx',
    sheet: 'Annual',
  });
  const headerRowIdx = requireRowIndex(rows, /^\d{4}-\d{2,4}$/, {
    labelCol: 3,
    to: growthRowIdx + 1,
    file: 'GDP_table.xlsx',
    sheet: 'Annual',
  });
  const headerRow = rows[headerRowIdx];
  const growthRow = rows[growthRowIdx];
  const growthUnit = requireUnit([growthRow], /GDP\s+Growth\s+Rate\s*\(\s*(?:%|percent)\s*\)/i);
  console.log(`  GDP rows: header@${headerRowIdx}, growth@${growthRowIdx}`);

  const existing = await readData('fiscal.json');
  const map = new Map();
  for (const e of existing.annual || []) map.set(e.year, { ...e });

  const seenYears = new Set();
  for (let col = 2; col < headerRow.length; col++) {
    const yrStr = (headerRow[col] || '').toString().trim();
    const m = yrStr.match(/^(\d{4})-(\d{2}|\d{4})(?:\s*\([PRF]\))?\s*$/);
    if (!m) {
      if (col >= 3 && (yrStr || (growthRow[col] !== null && growthRow[col] !== undefined))) throw new Error(`GDP source has missing/malformed fiscal period at column ${excelCell(col)}`);
      continue;
    }

    const fy = fiscalYearRange(yrStr, { context: `GDP period ${excelCell(col)}${headerRowIdx + 1}` });
    if (seenYears.has(fy)) throw new Error(`GDP source has duplicate fiscal year ${yrStr}`);
    seenYears.add(fy);
    if (fy < 2017 || fy > 2099) continue;

    const growth = cellNumber(rows, growthRowIdx, col);

    const key = `FY${fy}`;
    const entry = map.get(key) || { year: key };
    entry.gdpGrowth = round2(growth);
    entry.gdpGrowthStatus = publicationStatus(yrStr);
    entry.gdpGrowthSourceStatus = yrStr;
    entry.evidence = {
      ...entry.evidence,
      gdpGrowth: cellEvidence(rows, growthRowIdx, col, {
        unit: '%', sourceUnit: growthUnit.label,
        periodHeader: { row: headerRowIdx + 1, column: col + 1, cell: `${excelCell(col)}${headerRowIdx + 1}`, text: yrStr },
        unitHeader: { row: growthRowIdx + 1, column: growthUnit.column + 1, cell: `${excelCell(growthUnit.column)}${growthRowIdx + 1}`, text: growthUnit.label },
      }),
    };
    map.set(key, entry);
  }

  const annual = Array.from(map.values()).sort((a, b) => a.year.localeCompare(b.year));

  // Spread existing keys (e.g. publicFinance from API updates) to preserve them.
  // `dataSource` is deliberately preserved when already set — it usually names
  // the exact SBP EasyData series, which is more traceable than a generic label.
  await writeData('fiscal.json', {
    ...existing,
    annual,
    dataSource: existing.dataSource || 'SBP / PBS',
    lastUpdated: new Date().toISOString().split('T')[0],
    dataCoverage: `${annual[0]?.year} – ${annual.at(-1)?.year}`,
  });
  console.log(`  📊 ${annual.length} fiscal years with GDP growth data`);
  return annual.length;
}

// 4. BALANCE OF PAYMENTS (Balancepayment_BPM6.xls)
//
// A previous implementation read fixed cells (row 24, row 38, row 79 …) from
// BPM6_Summary and printed them as if they were authoritative, but it never
// wrote a JSON file and its labels were stale. It was removed.
//
// This replacement exists for one reason: SBP refreshes the BOP summary a
// month ahead of the detailed EBOPS services table, so the headline services
// aggregate for the newest month is published here first. Every row and
// column below is resolved by its printed label.
// ═══════════════════════════════════════════════════

async function updateBopServices() {
  console.log('\n🧾 Parsing BOP services aggregate (Balancepayment_BPM6.xls)...');

  const wb = await readExcel('Balancepayment_BPM6.xls');
  const rows = getSheet(wb, 'BPM6_Summary');
  requireUnit(rows, /Million\s+US\s*\$/i);
  const ctx = { file: 'Balancepayment_BPM6.xls', sheet: 'BPM6_Summary' };

  const headerIdx = requireRowIndex(rows, /^items$/i, { labelCol: 0, to: 20, ...ctx });
  const width = Math.max(rows[headerIdx]?.length || 0, rows[headerIdx + 1]?.length || 0);
  // "Jul-Jun" is a merged cell spanning the prior and current fiscal year.
  const periodRow = forwardFill(rows[headerIdx], width);
  const fyRow = rows[headerIdx + 1] || [];

  const maxFy = latestFiscalYear(fyRow);
  if (maxFy === null) {
    throw new Error('No fiscal year found in the BPM6_Summary header row');
  }

  const creditIdx = requireRowIndex(rows, /^exports of services$/i, { labelCol: 0, ...ctx });
  const debitIdx = requireRowIndex(rows, /^imports of services$/i, { labelCol: 0, ...ctx });
  const netIdx = requireRowIndex(rows, /^balance on trade in services$/i, { labelCol: 0, ...ctx });

  let cumulative = null;
  let latestMonth = null;
  for (let col = 1; col < width; col++) {
    if (parseFullFy(fyRow[col]) !== maxFy) continue;
    const period = (periodRow[col] || '').trim();
    if (!period) continue;
    const status = publicationStatus(fyRow[col]);

    const cumulativeEnd = fiscalPeriodEndIndex(period);
    if (cumulativeEnd != null) {
      if (!cumulative || cumulativeEnd > cumulative.endIndex) {
        cumulative = { col, period, status, endIndex: cumulativeEnd };
      }
      continue;
    }

    const monthIndex = FISCAL_MONTH_ORDER.indexOf(period.slice(0, 3).toLowerCase());
    if (monthIndex >= 0 && period.length <= 9 && !period.includes('-')) {
      if (!latestMonth || monthIndex > latestMonth.endIndex) {
        latestMonth = { col, period, status, endIndex: monthIndex };
      }
    }
  }

  const readBlock = (entry) => {
    if (!entry) return null;
    const num = (rowIdx) => round2(cellNumber(rows, rowIdx, entry.col));
    const credit = num(creditIdx);
    const debit = num(debitIdx);
    const net = num(netIdx);
    assertReconciled(`BOP services ${entry.period} FY${maxFy}`, net, [credit, -debit], { tolerance: 1 });
    return {
      period: entry.period,
      fiscalYear: `FY${maxFy}`,
      status: entry.status,
      credit,
      debit, net,
      coverage: /[-\u2013]/.test(entry.period) ? fiscalWindow(entry.period, maxFy)
        : { start: fiscalMonth(entry.period, maxFy), end: fiscalMonth(entry.period, maxFy) },
      evidence: { credit: cellEvidence(rows, creditIdx, entry.col), debit: cellEvidence(rows, debitIdx, entry.col), net: cellEvidence(rows, netIdx, entry.col) },
    };
  };

  const cumulativeBlock = readBlock(cumulative);
  const monthBlock = readBlock(latestMonth);
  if (!cumulativeBlock && !monthBlock) {
    throw new Error(`No FY${maxFy} services columns resolved in BPM6_Summary`);
  }

  const existing = await readJson('services.json');
  const bopSummary = {
    source: 'SBP Balance of Payments (BPM6) summary',
    sourceFile: 'Balancepayment_BPM6.xls',
    sourceSheet: 'BPM6_Summary',
    fiscalYear: `FY${maxFy}`,
    unit: 'US$ million',
    cumulative: cumulativeBlock,
    latestMonth: monthBlock,
    note: 'SBP publishes this aggregate before the detailed EBOPS table, so it can cover a later month than the category and IT breakdowns in this section.',
  };

  await writeJson('services.json', { ...existing, bopSummary });

  if (cumulativeBlock) {
    await recordProvenance('services.bop.cumulative', {
      label: 'Balance on trade in services, fiscal year to date',
      sourceKey: 'Balancepayment_BPM6.xls',
      sheet: 'BPM6_Summary',
      location: `"Balance on Trade in Services" row, "${cumulativeBlock.period} ${cumulativeBlock.fiscalYear}" column`,
      period: `${cumulativeBlock.period} ${cumulativeBlock.fiscalYear}`,
      status: cumulativeBlock.status,
      evidence: cumulativeBlock.evidence.net,
      unit: 'US$ million',
      value: cumulativeBlock.net,
    });
    console.log(`  📊 Services ${cumulativeBlock.period} FY${maxFy}: credit $${cumulativeBlock.credit}M, debit $${cumulativeBlock.debit}M, net $${cumulativeBlock.net}M`);
  }
  if (monthBlock) {
    await recordProvenance('services.bop.latestMonth', {
      label: 'Exports of services, latest published month',
      sourceKey: 'Balancepayment_BPM6.xls',
      sheet: 'BPM6_Summary',
      location: `"Exports of Services" row, "${monthBlock.period} ${monthBlock.fiscalYear}" column`,
      period: `${monthBlock.period} ${monthBlock.fiscalYear}`,
      status: monthBlock.status,
      evidence: monthBlock.evidence.credit,
      unit: 'US$ million',
      value: monthBlock.credit,
    });
    console.log(`  📊 Latest month ${monthBlock.period} FY${maxFy}: credit $${monthBlock.credit}M, net $${monthBlock.net}M`);
  }
  return bopSummary;
}

async function downloadExchangeRateArchive() {
  console.log('\n💱 Attempting to download exchange rate archive...');
  const urls = [
    'https://www.sbp.org.pk/assets/document/IBF_Arch.xls',
    'https://archive.sbp.org.pk/ecodata/IBF_Arch.xls',
  ];
  const filepath = resolve(RAW_DIR, 'IBF_Arch.xls');

  for (const url of urls) {
    try {
      const res = await fetch(url, {
        redirect: 'error',
        signal: AbortSignal.timeout(30000),
        headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' },
      });

      if (!res.ok) {
        console.log(`  ⚠️  HTTP ${res.status} from ${url}`);
        continue;
      }

      const buffer = Buffer.from(await res.arrayBuffer());
      if (buffer.length < 5000) {
        console.log(`  ⚠️  Response too small (${buffer.length} bytes) from ${url}`);
        continue;
      }

      const contentType = res.headers.get('content-type') || '';
      const header = buffer.subarray(0, 512).toString('utf8').trimStart().toLowerCase();
      if (contentType.toLowerCase().includes('text/html') || header.startsWith('<!doctype html') || !isLegacyExcel(buffer)) {
        console.log(`  ⚠️  Invalid XLS response from ${url}`);
        continue;
      }

      const { captureSourceArtifact } = await import('./lib/source-evidence.mjs');
      await captureSourceArtifact({
        sourceKey: 'IBF_Arch.xls', sourceUrl: officialUrl(url), responseUrl: officialUrl(res.url || url),
        body: buffer, contentType, filename: 'IBF_Arch.xls', parser: 'parse-sbp-excel.mjs',
      });
      await writeFile(filepath, buffer);
      console.log(`  ✅ Downloaded IBF_Arch.xls (${(buffer.length / 1024).toFixed(0)} KB)`);
      return true;
    } catch (err) {
      console.log(`  ⚠️  Download failed from ${url}: ${err.message}`);
    }
  }

  console.log('  ⚠️  No verified official exchange-rate source succeeded; existing archive preserved');
  return false;
}

async function updateExchangeRates() {
  const archivePath = resolve(RAW_DIR, 'IBF_Arch.xls');
  if (!(await fileExists(archivePath))) {
    throw new Error('No exchange-rate archive available');
  }

  console.log('\n💱 Parsing Exchange Rate Archive (IBF_Arch.xls)...');

  try {
    const wb = await readExcel('IBF_Arch.xls');
    const sheetName = wb.SheetNames.find(s =>
      /monthly.*avg.*pkr/i.test(s)
    );
    if (!sheetName) throw new Error('Monthly average PKR exchange-rate sheet not found');

    console.log(`  Using sheet: "${sheetName}"`);
    const rows = getSheet(wb, sheetName);

    // Structure: Row 7 = country names, Row 8 = currency names
    // Col 0 = month name ("January"), Col 1 = year (2026)
    // Col 5 = China (CNY), Col 22 = U.K. (GBP), Col 23 = U.S.A. (USD), Col 24 = EMU (EUR)
    // Detect columns from the first set of country/currency headers (cols 2-25)
    // The sheet has a duplicate set of columns starting around col 27 — ignore those.
    let usdCol = -1, eurCol = -1, gbpCol = -1, cnyCol = -1;

    requireUnit(rows, /^Pak Rupees per Currency Unit$/i);
    const countryIdx = requireRowIndex(rows, /^COUNTRY$/i, { to: 20 });
    const currencyIdx = requireRowIndex(rows, /^CURRENCY$/i, { to: 20 });
    const countryRow = rows[countryIdx];
    const currencyRow = rows[currencyIdx];
    const firstBlockEnd = countryRow.findIndex((value, c) => c > 1 && value === null && currencyRow[c] === null);
    if (firstBlockEnd < 0) throw new Error('Exchange-rate source has no explicit boundary between values and percentage changes');
    const maxCol = firstBlockEnd;
    for (let c = 0; c < maxCol; c++) {
      const country = (countryRow[c] || '').toString().toLowerCase().trim();
      const currency = (currencyRow[c] || '').toString().toLowerCase().trim();
      if (usdCol < 0 && (/u\.?s\.?a|united states/.test(country) || /u\.?s\.?\s*dollar/.test(currency))) usdCol = c;
      if (eurCol < 0 && (/emu|euro/i.test(country) || /^euro$/i.test(currency))) eurCol = c;
      if (gbpCol < 0 && (/u\.?k|united kingdom/i.test(country) || /pound sterling/i.test(currency))) gbpCol = c;
      if (cnyCol < 0 && (/china/i.test(country) || /yuan|renminbi/i.test(currency))) cnyCol = c;
    }

    if (usdCol < 0) {
      throw new Error('Could not identify required USD column');
    }

    console.log(`  Columns: USD=${usdCol} EUR=${eurCol} GBP=${gbpCol} CNY=${cnyCol}`);

    const MONTH_MAP = {
      january: '01', february: '02', march: '03', april: '04',
      may: '05', june: '06', july: '07', august: '08',
      september: '09', october: '10', november: '11', december: '12',
    };

    const monthly = [];
    // Data rows start after the header area; first monthly rows begin around row 9+
    // Early rows have range periods like "Aug 1947 to Jun 1949" — skip those
    for (let i = currencyIdx + 1; i < rows.length; i++) {
      const row = rows[i];
      if (!row) continue;

      const monthStr = (row[0] || '').toString().trim().toLowerCase();
      const yearVal = row[1];

      // Need a valid month name, a numeric year, and a numeric USD rate
      const mm = MONTH_MAP[monthStr] || ({
        jan: '01', feb: '02', mar: '03', apr: '04', may: '05', jun: '06',
        jul: '07', aug: '08', sep: '09', oct: '10', nov: '11', dec: '12',
      })[monthStr];
      if (!mm) {
        if (Number.isInteger(yearVal) && yearVal >= 2021) throw new Error(`Invalid exchange-rate month at row ${i + 1}: ${monthStr}`);
        continue;
      }
      if (!Number.isInteger(yearVal) || yearVal < 1900) throw new Error(`Invalid exchange-rate calendar year at row ${i + 1}`);

      const year = yearVal;
      if (year < 2021) continue;

      const date = calendarDate(`${year}-${mm}`, { monthly: true });
      const entry = { date, unit: 'PKR per currency unit', status: 'not-stated', evidence: {} };
      for (const [currency, col] of [['USD', usdCol], ['EUR', eurCol], ['GBP', gbpCol], ['CNY', cnyCol]]) {
        requireColumn(col, { currency });
        const value = cellNumber(rows, i, col, { required: currency === 'USD' });
        if (value !== null && value <= 0) throw new Error(`Invalid nonpositive ${currency} exchange rate at ${date}`);
        entry[currency] = round2(value);
        entry.evidence[currency] = cellEvidence(rows, i, col);
      }
      monthly.push(entry);
    }

    if (monthly.length === 0) {
      throw new Error('No exchange rate data extracted');
    }

    const existing = await readJson('exchange-rates.json');
    const history = new Map((existing.monthly || []).map(row => [row.date, row]));
    for (const row of uniqueObservations(monthly)) history.set(row.date, row);
    const preservedMonthly = uniqueObservations([...history.values()]);
    await writeJson('exchange-rates.json', {
      ...existing,
      monthly: preservedMonthly,
      dataSource: 'SBP',
      lastUpdated: new Date().toISOString().split('T')[0],
      dataCoverage: `${monthly[0]?.date} – ${monthly.at(-1)?.date}`,
    });
    console.log(`  📊 ${monthly.length} months (${monthly[0]?.date} → ${monthly.at(-1)?.date})`);
    return monthly.length;
  } catch (err) {
    throw new Error(`Exchange-rate parse failed: ${err.message}`, { cause: err });
  }
}

export function parseReserveTable(items, receipt) {
  if (!items.some(item => /\(\s*MILLION\s+US\s*\$\s*\)/i.test(item.text))) throw new Error('Reserves PDF has no printed unit');
  const headers = ['SBP', 'BANKS', 'TOTAL LIQUID'].map(label => {
    const match = items.find(item => String(item.text).trim().startsWith(label));
    if (!match) throw new Error(`Reserves PDF missing ${label} header`);
    return match;
  });
  const observations = [];
  for (const anchor of items) {
    const parsed = parseReservesDate(String(anchor.text).trim());
    if (!parsed || parsed.type === 'annual') continue;
    calendarDate(parsed.date, { monthly: parsed.type === 'monthly' });
    const cells = items.filter(item => String(item.text).trim() && item.page === anchor.page && item.x > anchor.x && Math.abs(item.y - anchor.y) <= 0.18).sort((a, b) => a.x - b.x);
    if (cells.length !== 3) throw new Error(`Reserves ${parsed.date}: requires all three published values, got ${cells.length}`);
    for (let i = 0; i < cells.length; i++) {
      if (Math.abs(cells[i].x - headers[i].x) > 3) throw new Error(`Reserves ${parsed.date}: numeric column does not match ${headers[i].text}`);
    }
    const [sbp, banks, total] = cells.map(cell => sourceNumber(cell.text, { context: `Reserves ${parsed.date}` }));
    assertReconciled(`Reserves SBP + banks ${parsed.date}`, total, [sbp, banks], { tolerance: 0.02 });
    observations.push({
      date: parsed.date, frequency: parsed.type, sbp: round2(sbp), banks: round2(banks), total: round2(total),
      unit: 'US$ million', status: publicationStatus(anchor.text),
      evidence: Object.fromEntries(['sbp', 'banks', 'total'].map((field, index) => [
        field, sourceEvidence(receipt, { page: cells[index].page, x: cells[index].x, y: cells[index].y, dateLabel: anchor.text, columnLabel: headers[index].text }, { rawValue: cells[index].text }),
      ])),
    });
  }
  if (!observations.length) throw new Error('Reserves PDF contains no observations');
  return uniqueObservations(observations);
}

// 7. SERVICES (dt.xls — EBOPS classification)
// ═══════════════════════════════════════════════════

function servicesSourceColumns(header) {
  const cumulative = [];
  const monthly = [];
  for (let column = 1; column < header.length; column++) {
    const raw = header[column];
    if (raw === null || raw === undefined || raw === '') continue;
    if (typeof raw === 'number') {
      excelDateToYYYYMM(raw);
      continue;
    }
    const label = String(raw).trim().replace(/\s*\([PRF]\)\s*$/i, '');
    const group = { credit: column, label, status: publicationStatus(raw), sourceStatus: raw };
    if (parseFullFy(raw) !== null) {
      const window = fiscalWindow(raw);
      cumulative.push({ ...group, fy: window.fiscalYear, window });
    } else {
      monthly.push({ ...group, month: calendarMonthLabel(raw) });
    }
  }
  uniqueObservations(cumulative.map(group => ({ date: `${group.window.start}/${group.window.end}` })));
  uniqueObservations(monthly.map(group => ({ date: group.month })));
  cumulative.sort((a, b) => a.window.end.localeCompare(b.window.end));
  const current = cumulative.at(-1);
  const prior = cumulative.find(group => group.fy === current?.fy - 1 && group.window.end.slice(5) === current.window.end.slice(5));
  monthly.sort((a, b) => a.month.localeCompare(b.month));
  if (!current || !prior || monthly.length > 2) throw new Error('EBOPS lacks unambiguous current/prior FY windows or has an unexpected monthly schema');
  return { current, prior, month1: monthly.length === 2 ? monthly[0] : null, month2: monthly.at(-1) ?? null };
}

async function updateServices({ loadWorkbook = readExcel, readData = readJson, writeData = writeJson } = {}) {
  console.log('\n💻 Parsing Services Data (dt.xls)...');

  const wb = await loadWorkbook('dt.xls');
  // Sheet name has trailing space: "EBOPS "
  const sheetName = wb.SheetNames.find(s => /EBOPS/i.test(s));
  if (!sheetName) throw new Error('EBOPS sheet not found. Available: ' + wb.SheetNames.join(', '));
  console.log(`  Using sheet: "${sheetName}"`);
  const rows = getSheet(wb, sheetName);

  // Resolve columns from header row (row 6). Each period group is
  // [Credit, Debit, Net]. No positional fallbacks — see note above.
  const hdr6 = rows[6] || [];
  requireUnit(rows, /Thousand\s+US\s*\$/i);
  const svcCols = servicesSourceColumns(hdr6);
  const resolveGroup = group => {
    if (!group) return null;
    const start = group.credit;
    const end = hdr6.findIndex((value, column) => column > start && value !== null && value !== '');
    const limit = end < 0 ? (rows[7] || []).length : end;
    for (const field of ['credit', 'debit', 'net']) {
      const matches = (rows[7] || []).map((value, column) => column >= start && column < limit && String(value).trim().toLowerCase() === field ? column : -1).filter(column => column >= 0);
      if (matches.length !== 1) throw new Error(`EBOPS missing/ambiguous ${field} subheader in ${hdr6[start]}`);
      group[field] = matches[0];
    }
    group.status = publicationStatus(hdr6[start]);
    return group;
  };
  resolveGroup(svcCols.current);
  resolveGroup(svcCols.prior);
  const months = [resolveGroup(svcCols.month1), resolveGroup(svcCols.month2)].filter(Boolean)
    .sort((a, b) => calendarMonthLabel(a.label).localeCompare(calendarMonthLabel(b.label)));
  svcCols.month1 = months.length === 2 ? months[0] : null;
  svcCols.month2 = months.at(-1) || null;
  const currentWindow = fiscalWindow(svcCols.current.label, svcCols.current.fy);
  const priorWindow = fiscalWindow(svcCols.prior.label, svcCols.prior.fy);
  if (currentWindow.end.slice(5) !== priorWindow.end.slice(5)) throw new Error('EBOPS comparison uses mismatched FY windows');
  const currentPeriodCols = { credit: svcCols.current.credit, debit: svcCols.current.debit, net: svcCols.current.net };
  const priorPeriodCols = { credit: svcCols.prior.credit, debit: svcCols.prior.debit, net: svcCols.prior.net };
  const month1Cols = svcCols.month1 ? { credit: svcCols.month1.credit, debit: svcCols.month1.debit, net: svcCols.month1.net } : null;
  const month2Cols = svcCols.month2 ? { credit: svcCols.month2.credit, debit: svcCols.month2.debit, net: svcCols.month2.net } : null;
  const month1Label = svcCols.month1?.label || '';
  const month2Label = svcCols.month2?.label || '';
  const currentPeriodLabel = svcCols.current.label;
  const priorPeriodLabel = svcCols.prior.label;
  const currentCum = svcCols.current;
  const priorCum = svcCols.prior;
  console.log(`  Services columns: current ${currentPeriodLabel} @${currentCum.credit}, prior ${priorPeriodLabel} @${priorCum.credit}`);


  const toM = (v) => round2(sourceNumber(v, { context: 'Required EBOPS source cell' }) / 1000);
  const evidenceFields = (row, mapping) => Object.fromEntries(Object.entries(mapping).map(([field, column]) => [field, cellEvidence(rows, row, column, { scale: 0.001, sourceUnit: 'US$ thousand', unit: 'US$ million' })]));

  // Every row below is located by its published label rather than by a fixed
  // index. SBP inserts and removes EBOPS lines between releases; a positional
  // read would silently attribute one service's numbers to another.
  const ebopsCtx = { file: 'dt.xls', sheet: sheetName };
  const rowByLabel = (pattern, opts = {}) => requireRowIndex(rows, pattern, { labelCol: 0, ...ebopsCtx, ...opts });

  const ROW = Object.fromEntries(
    Object.entries(EBOPS_ROW_PATTERNS).map(([key, pattern]) => [key, rowByLabel(pattern)]),
  );

  const categoryRows = [
    { row: ROW.transport, name: 'Transport' },
    { row: ROW.travel, name: 'Travel' },
    { row: ROW.insurance, name: 'Insurance & Pension' },
    { row: ROW.financial, name: 'Financial Services' },
    { row: ROW.ipCharges, name: 'IP Charges' },
    { row: ROW.itTelecom, name: 'IT & Telecom' },
    { row: ROW.otherBusiness, name: 'Other Business' },
    { row: ROW.personalCultural, name: 'Personal/Cultural' },
  ];

  const categories = [];
  for (const { row, name } of categoryRows) {
    const r = rows[row];
    if (!r) continue;
    categories.push({
      name,
      credit: toM(r[currentPeriodCols.credit]),
      debit: toM(r[currentPeriodCols.debit]),
      net: toM(r[currentPeriodCols.net]),
      priorCredit: toM(r[priorPeriodCols.credit]),
      priorDebit: toM(r[priorPeriodCols.debit]),
      priorNet: toM(r[priorPeriodCols.net]),
      period: currentPeriodLabel,
      status: currentCum.status, priorStatus: priorCum.status,
      evidence: evidenceFields(row, {
        credit: currentPeriodCols.credit, debit: currentPeriodCols.debit, net: currentPeriodCols.net,
        priorCredit: priorPeriodCols.credit, priorDebit: priorPeriodCols.debit, priorNet: priorPeriodCols.net,
      }),
    });
  }

  // IT sub-category breakdown
  const itBreakdown = [
    { row: ROW.telecom, name: 'Telecom' },
    { row: ROW.softwareConsultancy, name: 'Software Consultancy' },
    { row: ROW.softwareExportImport, name: 'Computer Software Export/Import' },
    { row: ROW.freelance, name: 'Freelance IT' },
    { row: ROW.informationServices, name: 'Information Services' },
  ];

  const itItems = [];
  for (const { row, name } of itBreakdown) {
    const r = rows[row];
    if (!r) continue;
    itItems.push({
      name,
      credit: toM(r[currentPeriodCols.credit]),
      priorCredit: toM(r[priorPeriodCols.credit]),
      status: currentCum.status, priorStatus: priorCum.status,
      evidence: evidenceFields(row, { credit: currentPeriodCols.credit, priorCredit: priorPeriodCols.credit }),
    });
  }

  // "Other Computer Services" = Computer services minus named subcategories
  const computerServicesRow = rows[ROW.computerServices];
  if (computerServicesRow) {
    const computerTotal = toM(computerServicesRow[currentPeriodCols.credit]);
    const namedSum = itItems.filter(i => ['Software Consultancy', 'Computer Software Export/Import', 'Freelance IT'].includes(i.name))
      .reduce((s, x) => s + x.credit, 0);
    const other = round2(computerTotal - namedSum);
    if (other < -0.02) throw new Error('EBOPS named computer-service components exceed published computer-services total');
    const priorComputerTotal = toM(computerServicesRow[priorPeriodCols.credit]);
    const priorNamedSum = itItems.filter(i => ['Software Consultancy', 'Computer Software Export/Import', 'Freelance IT'].includes(i.name))
      .reduce((s, x) => s + x.priorCredit, 0);
    const priorOther = round2(priorComputerTotal - priorNamedSum);
    if (priorOther < -0.02) throw new Error('EBOPS prior named computer-service components exceed published computer-services total');
    const namedItems = itItems.filter(item => ['Software Consultancy', 'Computer Software Export/Import', 'Freelance IT'].includes(item.name));
    itItems.push({
      name: 'Other Computer Services', credit: other < 0 ? null : other, priorCredit: priorOther < 0 ? null : priorOther,
      missingReasons: { credit: other < 0 ? 'source-rounding-discrepancy' : null, priorCredit: priorOther < 0 ? 'source-rounding-discrepancy' : null },
      status: currentCum.status, priorStatus: priorCum.status,
      evidence: {
        credit: derivedEvidence([cellEvidence(rows, ROW.computerServices, currentPeriodCols.credit, { scale: 0.001 }), ...namedItems.map(item => item.evidence.credit)], 'computer services less named components (all inputs in US$ million)'),
        priorCredit: derivedEvidence([cellEvidence(rows, ROW.computerServices, priorPeriodCols.credit, { scale: 0.001 }), ...namedItems.map(item => item.evidence.priorCredit)], 'computer services less named components (all inputs in US$ million)'),
      },
    });
  }

  // Total services row
  const totalRow = rows[ROW.totalServices];
  const totalCredit = toM(totalRow?.[currentPeriodCols.credit]);
  const totalDebit = toM(totalRow?.[currentPeriodCols.debit]);
  const totalNet = toM(totalRow?.[currentPeriodCols.net]);
  const totalCreditPrior = toM(totalRow?.[priorPeriodCols.credit]);

  // IT & Telecom aggregate
  const itRow = rows[ROW.itTelecom];
  const itCredit = toM(itRow?.[currentPeriodCols.credit]);
  const itNet = toM(itRow?.[currentPeriodCols.net]);
  const itCreditPrior = toM(itRow?.[priorPeriodCols.credit]);

  // Computer services
  const csCredit = toM(computerServicesRow?.[currentPeriodCols.credit]);

  // Monthly data for recent months (if available)
  const recentMonths = [];
  for (const [cols, label] of [[month1Cols, month1Label], [month2Cols, month2Label]]) {
    if (!cols) continue;
    const totalMo = toM(totalRow?.[cols.credit]);
    const itMo = toM(itRow?.[cols.credit]);
    recentMonths.push({
      month: label, totalCredit: totalMo, itCredit: itMo,
      status: publicationStatus(hdr6[cols.credit]),
      evidence: { totalCredit: cellEvidence(rows, ROW.totalServices, cols.credit, { scale: 0.001 }), itCredit: cellEvidence(rows, ROW.itTelecom, cols.credit, { scale: 0.001 }) },
    });
  }

  // ── IT & Freelance monthly snapshot (latest month, prior month, year-ago month,
  // plus FYTD current vs prior) per IT sub-component. The SBP EBOPS file publishes
  // only the latest two months + same-month-last-year + FYTD — every figure authentic.
  const labelToYearMonth = (label) => label ? calendarMonthLabel(label) : null;
  // The year-ago comparison month is stored as an Excel date serial in row 6.
  let yearAgoCol = null;
  let yearAgoMonth = null;
  for (let c = 0; c < hdr6.length; c++) {
    if (typeof hdr6[c] === 'number' && hdr6[c] > 40000 && hdr6[c] < 50000) {
      yearAgoCol = c;
      yearAgoMonth = excelDateToYYYYMM(hdr6[c]);
      break;
    }
  }

  const IT_COMPONENT_ROWS = [
    { row: ROW.itTelecom, name: 'IT & Telecom (total)', key: 'itTotal' },
    { row: ROW.telecom, name: 'Telecommunications', key: 'telecom' },
    { row: ROW.softwareConsultancy, name: 'Software Consultancy', key: 'softwareConsultancy' },
    { row: ROW.softwareExportImport, name: 'Computer Software Exports', key: 'softwareExports' },
    { row: ROW.freelance, name: 'Freelance IT', key: 'freelance' },
    { row: ROW.informationServices, name: 'Information Services', key: 'informationServices' },
  ];
  const components = IT_COMPONENT_ROWS.map(({ row, name, key }) => {
    const r = rows[row] || [];
    return {
      key,
      name,
      latest: month2Cols ? toM(r[month2Cols.credit]) : null,
      prev: month1Cols ? toM(r[month1Cols.credit]) : null,
      yearAgo: yearAgoCol != null ? toM(r[yearAgoCol]) : null,
      fytd: toM(r[currentPeriodCols.credit]),
      fytdPrior: toM(r[priorPeriodCols.credit]),
      statuses: {
        latest: svcCols.month2?.status ?? 'not-stated', prev: svcCols.month1?.status ?? 'not-stated',
        yearAgo: 'not-stated', fytd: currentCum.status, fytdPrior: priorCum.status,
      },
      evidence: evidenceFields(row, {
        ...(month2Cols ? { latest: month2Cols.credit } : {}),
        ...(month1Cols ? { prev: month1Cols.credit } : {}),
        ...(yearAgoCol !== null ? { yearAgo: yearAgoCol } : {}),
        fytd: currentPeriodCols.credit, fytdPrior: priorPeriodCols.credit,
      }),
    };
  });
  const fytdStem = (currentPeriodLabel.match(/Jul-\w+/) || ['Jul–latest'])[0].replace('-', '–');
  const itMonthly = {
    latestMonth: labelToYearMonth(month2Label),
    prevMonth: labelToYearMonth(month1Label),
    yearAgoMonth,
    fytdLabel: currentPeriodLabel,
    fytdPriorLabel: priorPeriodLabel,
    note: `SBP's EBOPS services file publishes the latest month, prior month, the same month a year earlier, and fiscal-year-to-date totals. Values are exports (credit) in US$ million. "Freelance IT" is SBP's own line for individual freelancer earnings; total IT & freelancing exports are captured within IT & Telecom (${fytdStem}).`,
    components,
  };

  // ── Accumulating monthly IT/freelance export series. SBP only exposes two months
  // at a time, so we persist them into a growing contiguous series across updates.
  const existingSvc = await readData('services.json');
  const seriesMap = new Map((existingSvc.monthlySeries || []).map((m) => [m.month, m]));
  const freelanceRow = rows[ROW.freelance] || [];
  for (const cols of [month1Cols, month2Cols]) {
    if (!cols) continue;
    const ym = labelToYearMonth(cols === month1Cols ? month1Label : month2Label);
    if (!ym) continue;
    const totalMo = toM(totalRow?.[cols.credit]);
    const itMo = toM(itRow?.[cols.credit]);
    const freelanceMo = toM(freelanceRow[cols.credit]);
    seriesMap.set(ym, {
      month: ym, totalCredit: totalMo, itCredit: itMo, freelanceCredit: freelanceMo,
      status: publicationStatus(hdr6[cols.credit]),
      evidence: {
        totalCredit: cellEvidence(rows, ROW.totalServices, cols.credit, { scale: 0.001 }),
        itCredit: cellEvidence(rows, ROW.itTelecom, cols.credit, { scale: 0.001 }),
        freelanceCredit: cellEvidence(rows, ROW.freelance, cols.credit, { scale: 0.001 }),
      },
    });
  }
  const monthlySeries = [...seriesMap.values()].sort((a, b) => a.month.localeCompare(b.month));
  const reconciliation = [];
  const categoryEnd = requireRowIndex(rows, /^12\.\s*Government goods and services/i);
  const topLevelRows = rows.slice(ROW.totalServices + 1, categoryEnd + 1).flatMap((row, index) => /^\d+\.\s+[A-Za-z]/.test(String(row[0])) ? [index + ROW.totalServices + 1] : []);
  if (topLevelRows.length !== 12) throw new Error('EBOPS source does not contain exactly 12 top-level service categories');
  for (const cols of [currentPeriodCols, priorPeriodCols, month1Cols, month2Cols].filter(Boolean)) {
    reconciliation.push(assertReconciled('EBOPS services credit less debit = net', cellNumber(rows, ROW.totalServices, cols.net),
      [cellNumber(rows, ROW.totalServices, cols.credit), -cellNumber(rows, ROW.totalServices, cols.debit)], { tolerance: 0.02 }));
    reconciliation.push(assertReconciled('EBOPS total services vs published top-level categories', cellNumber(rows, ROW.totalServices, cols.credit),
      topLevelRows.map(row => cellNumber(rows, row, cols.credit)), { tolerance: 0.02 }));
    reconciliation.push(assertReconciled('EBOPS IT total vs telecom + computer + information services', cellNumber(rows, ROW.itTelecom, cols.credit),
      [ROW.telecom, ROW.computerServices, ROW.informationServices].map(row => cellNumber(rows, row, cols.credit)), { tolerance: 0.02 }));
  }
  reconciliation.push(...await reconcileBopObservations(monthlySeries.map(row => ({ ...row, date: row.month })), [
    ['totalCredit', /^exports of services$/i, 'BOP-services-credit'],
  ], { loadWorkbook }));
  const bopWb = await loadWorkbook('Balancepayment_BPM6.xls');
  const bopRows = getSheet(bopWb, 'BPM6_Summary');
  for (const block of bopSourceBlocks(bopRows, /^exports of services$/i, 'BOP-services-credit')) {
    reconciliation.push(compareEquivalent('EBOPS cumulative services vs independent BOP total', {
      ...currentWindow, basis: block.basis, unit: block.unit, value: totalCredit,
      evidence: cellEvidence(rows, ROW.totalServices, currentPeriodCols.credit, { scale: 0.001 }),
    }, block, { tolerance: 1 }));
  }

  const servicesData = {
    ...existingSvc,
    categories,
    itBreakdown: itItems,
    summary: {
      totalServicesCredit: totalCredit,
      totalServicesDebit: totalDebit,
      totalServicesNet: totalNet,
      itTelecomCredit: itCredit,
      itTelecomNet: itNet,
      computerServicesCredit: csCredit,
      period: currentPeriodLabel,
      status: currentCum.status,
      evidence: {
        totalServicesCredit: cellEvidence(rows, ROW.totalServices, currentPeriodCols.credit, { scale: 0.001 }),
        totalServicesDebit: cellEvidence(rows, ROW.totalServices, currentPeriodCols.debit, { scale: 0.001 }),
        totalServicesNet: cellEvidence(rows, ROW.totalServices, currentPeriodCols.net, { scale: 0.001 }),
        itTelecomCredit: cellEvidence(rows, ROW.itTelecom, currentPeriodCols.credit, { scale: 0.001 }),
        itTelecomNet: cellEvidence(rows, ROW.itTelecom, currentPeriodCols.net, { scale: 0.001 }),
        computerServicesCredit: cellEvidence(rows, ROW.computerServices, currentPeriodCols.credit, { scale: 0.001 }),
      },
    },
    comparison: {
      // `current`/`prior` are the stable keys. The FY-named aliases are kept for
      // backwards compatibility but must never be relied on — they were a latent
      // bug waiting for the next fiscal year to roll over.
      current: { totalCredit: totalCredit, itCredit: itCredit, fy: currentCum.fy, status: currentCum.status, evidence: { totalCredit: cellEvidence(rows, ROW.totalServices, currentPeriodCols.credit, { scale: 0.001 }), itCredit: cellEvidence(rows, ROW.itTelecom, currentPeriodCols.credit, { scale: 0.001 }) } },
      prior: { totalCredit: totalCreditPrior, itCredit: itCreditPrior, fy: priorCum.fy, status: priorCum.status, evidence: { totalCredit: cellEvidence(rows, ROW.totalServices, priorPeriodCols.credit, { scale: 0.001 }), itCredit: cellEvidence(rows, ROW.itTelecom, priorPeriodCols.credit, { scale: 0.001 }) } },
      [`fy${String(priorCum.fy).slice(-2)}`]: { totalCredit: totalCreditPrior, itCredit: itCreditPrior, status: priorCum.status, evidence: { totalCredit: cellEvidence(rows, ROW.totalServices, priorPeriodCols.credit, { scale: 0.001 }), itCredit: cellEvidence(rows, ROW.itTelecom, priorPeriodCols.credit, { scale: 0.001 }) } },
      [`fy${String(currentCum.fy).slice(-2)}`]: { totalCredit: totalCredit, itCredit: itCredit, status: currentCum.status, evidence: { totalCredit: cellEvidence(rows, ROW.totalServices, currentPeriodCols.credit, { scale: 0.001 }), itCredit: cellEvidence(rows, ROW.itTelecom, currentPeriodCols.credit, { scale: 0.001 }) } },
      period: currentPeriodLabel.replace(/FY\d{2}/, '').trim().replace(/,$/, ''),
      currentLabel: `FY${String(currentCum.fy).slice(-2)}`,
      priorLabel: `FY${String(priorCum.fy).slice(-2)}`,
    },
    recentMonths,
    itMonthly,
    monthlySeries,
    reconciliation,
    dataSource: 'SBP',
    lastUpdated: new Date().toISOString().slice(0, 10),
    dataCoverage: currentPeriodLabel,
  };

  await writeData('services.json', servicesData);

  console.log(`  📊 ${categories.length} service categories, ${itItems.length} IT sub-categories`);
  console.log(`     Total Services Credit: $${totalCredit}M, IT&Telecom: $${itCredit}M`);
  if (recentMonths.length > 0) console.log(`     Recent months: ${recentMonths.map(m => `${m.month}: $${m.totalCredit}M`).join(', ')}`);
  console.log(`     IT monthly snapshot: latest ${itMonthly.latestMonth || '?'}, ${components.length} components; accumulating series ${monthlySeries.length} month(s)`);
  return servicesData;
}

async function updateServicesHeadline() {
  console.log('\n💻 Parsing Services Headline (ExportsImports-Goods.pdf)...');

  const { receipt } = await sourceReceipt('ExportsImports-Goods.pdf');
  const pdfItems = await parsePdfTextItems(resolve(RAW_DIR, 'ExportsImports-Goods.pdf'));
  const headline = parseServicesHeadline(pdfItems, { receipt });
  const existing = await readJson('services.json');
  const detail = existing.itMonthly || {};
  const detailLatestMonth = detail.detailLatestMonth || detail.latestMonth || null;
  const detailYearAgoMonth = detail.detailYearAgoMonth || detail.yearAgoMonth || null;
  const detailFytdLabel = detail.detailFytdLabel || detail.fytdLabel || null;
  const detailFytdPriorLabel = detail.detailFytdPriorLabel || detail.fytdPriorLabel || null;
  const components = (detail.components || []).map((component) => ({
    ...component,
    latestMonth: component.latestMonth || detailLatestMonth,
    yearAgoMonth: component.yearAgoMonth || detailYearAgoMonth,
    fytdLabel: component.fytdLabel || detailFytdLabel,
    fytdPriorLabel: component.fytdPriorLabel || detailFytdPriorLabel,
  }));
  if (!components.some((component) => component.key === 'itTotal')) {
    throw new Error('Services detail is missing the IT total component');
  }

  const seriesMap = new Map((existing.monthlySeries || []).map((row) => [row.month, row]));
  const previousLatest = seriesMap.get(headline.latestMonth);
  const reconciliation = [...headline.reconciliation];
  if (previousLatest?.evidence?.itCredit) {
    reconciliation.push(compareEquivalent('Headline IT vs independent EBOPS same month', {
      start: headline.latestMonth, end: headline.latestMonth, basis: 'BOP-IT-credit', unit: headline.unit, value: headline.latest, evidence: headline.evidence.latest,
    }, {
      start: previousLatest.month, end: previousLatest.month, basis: 'BOP-IT-credit', unit: headline.unit, value: previousLatest.itCredit, evidence: previousLatest.evidence.itCredit,
    }));
  }
  reconciliation.push(...await reconcileBopObservations([{
    date: headline.latestMonth, totalCredit: headline.totalServicesLatest, evidence: { totalCredit: headline.evidence.totalServicesLatest },
  }], [['totalCredit', /^exports of services$/i, 'BOP-services-credit']]));
  if (detail.fytdLabel && existing.comparison?.current) {
    reconciliation.push(compareEquivalent('Headline IT cumulative vs independent EBOPS', {
      ...headline.period, basis: 'BOP-IT-credit', unit: headline.unit, value: headline.fytd, evidence: headline.evidence.fytd,
    }, {
      ...fiscalWindow(detail.fytdLabel, existing.comparison.current.fy), basis: 'BOP-IT-credit', unit: headline.unit,
      value: existing.comparison.current.itCredit, evidence: existing.comparison.current.evidence?.itCredit,
    }));
  }
  seriesMap.set(headline.latestMonth, {
    ...previousLatest,
    month: headline.latestMonth,
    totalCredit: headline.totalServicesLatest,
    itCredit: headline.latest,
    freelanceCredit: seriesMap.get(headline.latestMonth)?.freelanceCredit ?? null,
    status: headline.status,
    statuses: { ...previousLatest?.statuses, totalCredit: headline.statuses.totalServicesLatest, itCredit: headline.statuses.latest, freelanceCredit: previousLatest?.status ?? 'not-stated' },
    evidence: { ...previousLatest?.evidence, totalCredit: headline.evidence.totalServicesLatest, itCredit: headline.evidence.latest },
  });

  const recentMonths = [...seriesMap.values()]
    .sort((a, b) => a.month.localeCompare(b.month))
    .slice(-4)
    .map((row) => {
      const [year, month] = String(row.month).split('-');
      const names = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
      const label = month ? `${names[Number(month) - 1]}-${String(year).slice(-2)}` : row.month;
      return {
        month: label,
        totalCredit: row.totalCredit,
        itCredit: row.itCredit,
        status: row.status ?? 'not-stated',
        evidence: row.evidence,
      };
    });

  const updated = {
    ...existing,
    recentMonths,
    itHeadline: {
      ...headline,
      source: 'State Bank of Pakistan',
      sourceFile: 'ExportsImports-Goods.pdf',
      sourceUrl: 'https://www.sbp.org.pk/assets/document/ExportsImports-Goods.pdf',
      reconciliation,
    },
    itMonthly: {
      ...detail,
      detailLatestMonth,
      detailYearAgoMonth,
      detailFytdLabel,
      detailFytdPriorLabel,
      note: `SBP's headline services table publishes total IT & Telecom through ${headline.latestMonth}, while the detailed EBOPS workbook currently publishes subcomponents through ${detailLatestMonth}. Values are exports (credit) in US$ million; component cards retain their own coverage dates.`,
      components,
    },
    monthlySeries: [...seriesMap.values()].sort((a, b) => a.month.localeCompare(b.month)),
    headlineCoverage: headline.latestMonth,
    lastUpdated: new Date().toISOString().slice(0, 10),
  };

  await writeJson('services.json', updated);
  await recordProvenance('services.itTelecom.headline', {
    label: 'Telecommunications, computer and information services exports (credit), latest month',
    value: headline.latest,
    unit: 'US$ million',
    sourceKey: 'ExportsImports-Goods.pdf',
    location: `Row 9 "Telecommunications, Computer, and Information Services", ${headline.latestMonth} column`,
    status: headline.status, evidence: headline.evidence.latest,
  });
  console.log(`  📊 IT & Telecom headline: $${headline.latest}M in ${headline.latestMonth}; detailed EBOPS through ${detailLatestMonth}`);
  return headline;
}

// 8. TRADE BY COUNTRY (Export/Import by country files)
// ═══════════════════════════════════════════════════

// Column classification for the by-country workbooks now lives in
// scripts/lib/sbp-resolvers.mjs so it can be unit-tested against captured
// header fixtures, including synthetic future fiscal years.

// Read a by-country export/import workbook into a Map keyed by clean country
// name, capturing the snapshot columns (values converted thousand-USD → $M).
async function parseCountryTradeFile(filename, sheetRegex) {
  const wb = await readExcel(filename);
  const sheetName = wb.SheetNames.find((s) => sheetRegex.test(s));
  if (!sheetName) throw new Error(`${filename}: required country sheet ${sheetRegex} not found`);
  const rows = getSheet(wb, sheetName);
  requireUnit(rows, /Thousand\s+US\s*(?:Dollar|\$)/i);
  const cols = countryTradeColumns(rows);

  const cell = (row, c, required = false) => {
    if (c === undefined) return null;
    return round2(cellNumber(rows, row, c, { required, divisor: 1000 }));
  };

  // Fail loudly rather than silently reading whatever sits in a guessed column.
  requireColumn(cols.fytdCur, { file: filename, sheet: sheetName, want: `Jul-* FY${cols.curFY} fiscal-year-to-date column` });
  requireColumn(cols.fytdPrior, { file: filename, sheet: sheetName, want: `Jul-* FY${cols.priorFY} fiscal-year-to-date column` });
  requireColumn(cols.latest, { file: filename, sheet: sheetName, want: `latest provisional month column for FY${cols.curFY}` });

  const map = new Map();
  for (let i = 6; i < rows.length; i++) {
    const row = rows[i];
    if (!row) continue;
    let country = (row[0] || '').toString().trim();
    if (/^I\.\s*(?:Export Receipts|Import Payments).*Banks/i.test(country)) break;
    if (!country || /^total|^grand|^sub|^all/i.test(country)) continue;
    if (/^\d+$/.test(country)) continue;
    if (/^[IVX]+\./i.test(country)) continue;
    if (/receipts|payments|through\s+banks|memo|of which/i.test(country)) continue;
    country = country.replace(/\s*\*+$/, '').trim();
    if (map.has(country)) throw new Error(`${filename}: duplicate country ${country}`);
    const mapping = { latest: cols.latest, prev: cols.prev, yearAgo: cols.yearAgo, fytd: cols.fytdCur, fytdPrior: cols.fytdPrior };
    map.set(country, {
      latest: cell(i, cols.latest), prev: cell(i, cols.prev), yearAgo: cell(i, cols.yearAgo),
      fytd: cell(i, cols.fytdCur, true), fytdPrior: cell(i, cols.fytdPrior, true),
      statuses: Object.fromEntries(Object.entries(mapping).map(([field, c]) => [field, c === undefined ? 'not-stated' : publicationStatus(`${forwardFill(rows[4])[c]} ${rows[5][c]}`)])),
      missingReasons: Object.fromEntries(Object.entries(mapping).map(([field, c]) => [field, c === undefined ? 'period-not-published' : sourceMissingReason(rows[i][c])])),
      evidence: Object.fromEntries(Object.entries(mapping).filter(([, c]) => c !== undefined).map(([field, c]) => [field, cellEvidence(rows, i, c, { scale: 0.001 })])),
    });
  }
  if (!map.size) throw new Error(`${filename}: no country observations`);
  const bankTotalRow = requireRowIndex(rows, /^I\.\s*(?:Export Receipts|Import Payments).*Banks/i);
  const reconciliation = [];
  for (const [field, c] of Object.entries({ latest: cols.latest, prev: cols.prev, yearAgo: cols.yearAgo, fytd: cols.fytdCur, fytdPrior: cols.fytdPrior })) {
    if (c === undefined) continue;
    const components = [...map.values()].map(row => row[field]);
    if (components.some(value => value === null)) {
      reconciliation.push({ check: `${filename} ${field} country components`, result: 'not-comparable', reason: 'missing-components' });
      continue;
    }
    reconciliation.push(assertReconciled(`${filename} ${field} country components = banks total`, cell(bankTotalRow, c, true), components, { tolerance: map.size * 0.005 + 0.01 }));
  }
  const bopTotalRow = requireRowIndex(rows, /^Total (?:Export|Imports?) as per BOP/i);
  const observations = [];
  for (const field of ['latest', 'prev', 'yearAgo']) {
    const c = cols[field];
    if (c === undefined) continue;
    const date = cols[`${field}Month`];
    observations.push({
      date, value: cell(bankTotalRow, c, true), bopValue: cell(bopTotalRow, c, true),
      evidence: { value: cellEvidence(rows, bankTotalRow, c, { scale: 0.001 }), bopValue: cellEvidence(rows, bopTotalRow, c, { scale: 0.001 }) },
    });
  }
  return { sheetName, cols, map, reconciliation, observations };
}

export function countryTradeColumns(rows) {
  if (!/^Country$/i.test(String(rows[4]?.[0]).trim())) throw new Error('Country trade table is missing Country header');
  const periods = forwardFill(rows[4]);
  const columns = rows[5].flatMap((value, column) => {
    if (value === null || value === '') return [];
    const fy = parseFullFy(value);
    if (fy === null) throw new Error(`Country trade has malformed fiscal year ${value}`);
    const rawPeriod = periods[column].replace(/\s*\([PRF]\)\s*$/i, '').trim();
    if (/[-\u2013]/.test(rawPeriod)) return [{ column, fy, period: rawPeriod, window: fiscalWindow(rawPeriod, fy) }];
    return [{ column, fy, period: rawPeriod, month: fiscalMonth(rawPeriod, fy) }];
  });
  const currentFy = Math.max(...columns.map(column => column.fy));
  const monthly = columns.filter(column => column.fy === currentFy && column.month).sort((a, b) => a.month.localeCompare(b.month));
  const cumulative = columns.filter(column => column.fy === currentFy && column.window).sort((a, b) => a.window.end.localeCompare(b.window.end)).at(-1);
  const prior = columns.find(column => column.fy === currentFy - 1 && column.window && column.period === cumulative?.period);
  const latest = monthly.at(-1);
  const prev = monthly.at(-2);
  const yearAgo = columns.find(column => column.month === (latest && `${Number(latest.month.slice(0, 4)) - 1}${latest.month.slice(4)}`));
  if (!latest || !cumulative || !prior) throw new Error('Country trade lacks latest month or comparable cumulative FY pair');
  uniqueObservations(monthly.map(column => ({ date: column.month })));
  return {
    curFY: currentFy % 100, priorFY: (currentFy - 1) % 100, curFullFY: currentFy,
    latest: latest.column, latestMonth: latest.month,
    ...(prev ? { prev: prev.column, prevMonth: prev.month } : {}),
    ...(yearAgo ? { yearAgo: yearAgo.column, yearAgoMonth: yearAgo.month } : {}),
    fytdCur: cumulative.column, fytdPrior: prior.column, fytdLabel: cumulative.period,
    coverage: { latest: { start: latest.month, end: latest.month }, prev: prev ? { start: prev.month, end: prev.month } : null, yearAgo: yearAgo ? { start: yearAgo.month, end: yearAgo.month } : null, fytd: cumulative.window, fytdPrior: prior.window },
  };
}

async function updateTradeCountries() {
  console.log('\n🌍 Parsing Trade by Country...');

  const countryPeriodLabel = (headerPeriods, headerYears, col) => {
    let period = (headerPeriods[col] || '').toString().trim();
    for (let c = col - 1; !period && c >= 0; c--) {
      period = (headerPeriods[c] || '').toString().trim();
    }
    const year = (headerYears[col] || '').toString().trim();
    return [period, year].filter(Boolean).join(' ');
  };

  // --- Exports by Country ---
  console.log('  📋 Export destinations (Export_Receipts_by_all_Countries.xls)...');
  const wbExp = await readExcel('Export_Receipts_by_all_Countries.xls');
  const expSheetName = wbExp.SheetNames.find(s => /Exp.*Acount/i.test(s));
  if (!expSheetName) throw new Error('Export country sheet not found');
  console.log(`  Using sheet: "${expSheetName}"`);
  const expRows = getSheet(wbExp, expSheetName);
  requireUnit(expRows, /Thousand\s+US\s*(?:Dollar|\$)/i);

  // The current fiscal-year-to-date column is resolved from the sheet's own
  // headers (row 4 = period, row 5 = fiscal year). Never hardcode a fiscal year
  // and never fall back to a fixed column index — when SBP shifts the layout we
  // must fail loudly rather than publish whatever happens to sit there.
  const expHeader4 = expRows[4] || [];
  const expHeader5 = expRows[5] || [];
  const expCols = countryTradeColumns(expRows);
  const expCol = requireColumn(expCols.fytdCur, {
    file: 'Export_Receipts_by_all_Countries.xls',
    sheet: expSheetName,
    want: `Jul-* FY${expCols.curFY} fiscal-year-to-date export column`,
    headerPeriods: expHeader4.filter(Boolean).join(' | '),
    headerYears: expHeader5.filter(Boolean).join(' | '),
  });
  console.log(`  Export data column: ${expCol} (FY${expCols.curFY} ${expCols.fytdLabel || ''})`);
  const exportCountryPeriod = countryPeriodLabel(expHeader4, expHeader5, expCol);

  const exportCountries = [];
  for (let i = 6; i < expRows.length; i++) {
    const row = expRows[i];
    if (!row) continue;
    const country = (row[0] || '').toString().trim();
    if (/^I\.\s*Export Receipts.*Banks/i.test(country)) break;
    if (!country || /^total|^grand|^sub|^all|^others?\s*$/i.test(country)) continue;
    if (/^\d+$/.test(country)) continue;
    // Skip section headers like "I. Export Receipts through Banks", "II. ..."
    if (/^[IVX]+\./i.test(country)) continue;
    if (/receipts|payments|through\s+banks|memo|of which/i.test(country)) continue;

    const val = cellNumber(expRows, i, expCol);

    // Convert from thousands to millions
    exportCountries.push({
      country: country.replace(/\s*\*+$/, '').trim(),
      value: round2(val / 1000),
      flag: getFlag(country.replace(/\s*\*+$/, '').trim()),
      status: publicationStatus(`${forwardFill(expRows[4])[expCol]} ${expRows[5][expCol]}`),
      evidence: { value: cellEvidence(expRows, i, expCol, { scale: 0.001 }) },
    });
  }

  exportCountries.sort((a, b) => b.value - a.value);
  const topExportCountries = exportCountries.slice(0, 15);

  // --- Imports by Country ---
  console.log('  📋 Import sources (Import-Payments-by-All-Countries.xlsx)...');
  const wbImp = await readExcel('Import-Payments-by-All-Countries.xlsx');
  const impSheetName = wbImp.SheetNames.find(s => /Import/i.test(s));
  if (!impSheetName) throw new Error('Import country sheet not found');
  console.log(`  Using sheet: "${impSheetName}"`);
  const impRows = getSheet(wbImp, impSheetName);
  requireUnit(impRows, /Thousand\s+US\s*(?:Dollar|\$)/i);

  const impHeader4 = impRows[4] || [];
  const impHeader5 = impRows[5] || [];
  const impCols = countryTradeColumns(impRows);
  const impCol = requireColumn(impCols.fytdCur, {
    file: 'Import-Payments-by-All-Countries.xlsx',
    sheet: impSheetName,
    want: `Jul-* FY${impCols.curFY} fiscal-year-to-date import column`,
    headerPeriods: impHeader4.filter(Boolean).join(' | '),
    headerYears: impHeader5.filter(Boolean).join(' | '),
  });
  console.log(`  Import data column: ${impCol} (FY${impCols.curFY} ${impCols.fytdLabel || ''})`);
  const importCountryPeriod = countryPeriodLabel(impHeader4, impHeader5, impCol);

  const importCountries = [];
  for (let i = 6; i < impRows.length; i++) {
    const row = impRows[i];
    if (!row) continue;
    const country = (row[0] || '').toString().trim();
    if (/^I\.\s*Import Payments.*Banks/i.test(country)) break;
    if (!country || /^total|^grand|^sub|^all|^others?\s*$/i.test(country)) continue;
    if (/^\d+$/.test(country)) continue;
    if (/^[IVX]+\./i.test(country)) continue;
    if (/receipts|payments|through\s+banks|memo|of which/i.test(country)) continue;

    const val = cellNumber(impRows, i, impCol);

    importCountries.push({
      country: country.replace(/\s*\*+$/, '').trim(),
      value: round2(val / 1000),
      flag: getFlag(country.replace(/\s*\*+$/, '').trim()),
      status: publicationStatus(`${forwardFill(impRows[4])[impCol]} ${impRows[5][impCol]}`),
      evidence: { value: cellEvidence(impRows, i, impCol, { scale: 0.001 }) },
    });
  }

  importCountries.sort((a, b) => b.value - a.value);
  const topImportCountries = importCountries.slice(0, 15);

  // --- Per-country monthly snapshot (latest month, prior month, year-ago month,
  // FYTD current & prior) for the most important trade partners. The SBP
  // by-country files only publish these few points per country (no long monthly
  // series), so this is a snapshot + YoY/MoM/FYTD view — every figure authentic.
  const expParsed = await parseCountryTradeFile('Export_Receipts_by_all_Countries.xls', /Exp.*Acount/i);
  const impParsed = await parseCountryTradeFile('Import-Payments-by-All-Countries.xlsx', /Import/i);

  // Normalise country names so export & import rows line up and split sub-rows
  // (e.g. UAE is listed as "Dubai" + "Abu Dhabi") collapse into one partner.
  const normName = (s) => s.toLowerCase().replace(/\.\s*/g, '').replace(/[^a-z0-9 ]/g, ' ').replace(/\s+/g, ' ').trim();
  const canonical = (raw) => {
    const n = normName(raw);
    if (/^u ?a ?e|^uae|abu ?dhabi|^dubai|sharjah/.test(n)) return { key: 'uae', display: 'U.A.E.' };
    if (/^u ?s ?a|united states|^usa/.test(n)) return { key: 'usa', display: 'United States' };
    if (/^u ?k|united kingdom/.test(n)) return { key: 'uk', display: 'United Kingdom' };
    if (/netherland|holland/.test(n)) return { key: 'netherlands', display: 'Netherlands' };
    if (/saudi/.test(n)) return { key: 'saudi arabia', display: 'Saudi Arabia' };
    if (/south korea|korea, rep|rep.*korea|^korea/.test(n) && !/north/.test(n)) return { key: 'south korea', display: 'South Korea' };
    // Default: strip parentheticals for display
    const display = raw.replace(/\s*\(.*?\)\s*/g, ' ').replace(/\s+/g, ' ').trim();
    return { key: n, display };
  };
  const FIELDS = ['latest', 'prev', 'yearAgo', 'fytd', 'fytdPrior'];
  const blank = () => ({
    latest: null, prev: null, yearAgo: null, fytd: null, fytdPrior: null, evidence: {}, statuses: {},
    missingReasons: Object.fromEntries(FIELDS.map(field => [field, 'country-not-listed'])), memberCount: 0,
  });
  const addSnap = (target, snap) => {
    if (!snap) return;
    for (const f of FIELDS) {
      target[f] = target.memberCount === 0 ? snap[f] : round2(sumComplete([target[f], snap[f]]));
      target.missingReasons[f] = target[f] === null ? (target.memberCount === 0 ? snap.missingReasons[f] : 'missing-component') : null;
      if (snap.evidence[f]) {
        const inputs = target.evidence[f]?.inputs ?? [];
        target.evidence[f] = derivedEvidence([...inputs, snap.evidence[f]], 'sum of disjoint source countries/territories');
      }
      target.statuses[f] = target.memberCount === 0 ? snap.statuses[f] : target.statuses[f] === snap.statuses[f] ? target.statuses[f] : 'mixed';
    }
    target.memberCount++;
  };
  const merged = new Map();
  const collect = (map, kind) => {
    const parentNames = new Set();
    for (const country of map.keys()) {
      const { key } = canonical(country);
      if (key === 'uae' && /^(?:u\.?\s*a\.?\s*e\.?|united arab emirates)$/i.test(country.trim())) parentNames.add(key);
    }
    for (const [country, snap] of map) {
      const { key, display } = canonical(country);
      if (!key) continue;
      if (parentNames.has(key) && /dubai|abu\s*dhabi|sharjah/i.test(country)) {
        throw new Error('Country trade contains both UAE total and nested territories; cannot prove disjoint coverage');
      }
      if (!merged.has(key)) merged.set(key, { name: display, exports: blank(), imports: blank() });
      addSnap(merged.get(key)[kind], snap);
    }
  };
  collect(expParsed.map, 'exports');
  collect(impParsed.map, 'imports');

  // Rank by total trade engagement (export FYTD + import FYTD), keep the leaders.
  const sameFytdCoverage = expParsed.cols.coverage.fytd.start === impParsed.cols.coverage.fytd.start
    && expParsed.cols.coverage.fytd.end === impParsed.cols.coverage.fytd.end;
  const ranked = [...merged.values()]
    .map((e) => ({
      ...e, totalTrade: sameFytdCoverage ? sumComplete([e.exports.fytd, e.imports.fytd]) : null,
      rankingValue: sameFytdCoverage ? sumComplete([e.exports.fytd, e.imports.fytd]) : e.exports.fytd,
    }))
    .filter((e) => e.rankingValue !== null)
    .sort((a, b) => b.rankingValue - a.rankingValue);

  // Always include key remittance/strategic partners even if outside the top trade list.
  const MUST_INCLUDE = ['saudi arabia', 'usa', 'uk', 'uae', 'china'];
  const importantKeys = new Set(ranked.slice(0, 18).map((e) => canonical(e.name).key));
  for (const k of MUST_INCLUDE) importantKeys.add(k);
  const countries = ranked
    .filter((e) => importantKeys.has(canonical(e.name).key))
    .map((e) => {
      const snapshot = (snap, coverage) => ({
        ...Object.fromEntries(FIELDS.map(field => [field, snap[field]])),
        statuses: snap.statuses, missingReasons: snap.missingReasons, evidence: snap.evidence, coverage, unit: 'US$ million',
      });
      return {
        country: e.name, flag: getFlag(e.name),
        exports: snapshot(e.exports, expParsed.cols.coverage),
        imports: snapshot(e.imports, impParsed.cols.coverage),
      };
    });

  const c = expParsed.cols;
  const fytdStem = (c.fytdLabel || 'Jul–latest').replace(/-/g, '–');
  const fytdLabel = `${fytdStem} FY${c.curFY}`;
  const fytdPriorLabel = `${fytdStem} FY${c.priorFY}`;
  const countryMonthly = {
    latestMonth: c.latestMonth || null,
    prevMonth: c.prevMonth || null,
    yearAgoMonth: c.yearAgoMonth || null,
    fytdLabel,
    fytdPriorLabel,
    note: 'SBP by-country files publish the latest month, prior month, same month a year earlier, and fiscal-year-to-date totals per country. Values in US$ million.',
    countries,
    exportCoverage: expParsed.cols.coverage,
    importCoverage: impParsed.cols.coverage,
    rankingBasis: sameFytdCoverage ? 'combined-trade-same-FY-window' : 'exports-only-import-window-differs',
  };

  // Update trade.json — merge with existing data
  const existing = await readJson('trade.json');
  existing.topExportCountries = topExportCountries;
  existing.topImportCountries = topImportCountries;
  existing.exportCountryPeriod = exportCountryPeriod || null;
  existing.importCountryPeriod = importCountryPeriod || null;
  existing.countryMonthly = countryMonthly;
  const monthlyTrade = existing.monthly || [];
  const reconciliation = [...expParsed.reconciliation, ...impParsed.reconciliation];
  for (const [parsed, field] of [[expParsed, 'exports'], [impParsed, 'imports']]) {
    for (const observation of parsed.observations) {
      const archiveRow = monthlyTrade.find(row => row.date === observation.date);
      if (!archiveRow) continue;
      reconciliation.push(compareEquivalent(`Country workbook independent BOP ${field} total vs archive`, {
        start: observation.date, end: observation.date, basis: `BOP-goods-${field}`, unit: 'US$ million',
        value: observation.bopValue, evidence: observation.evidence.bopValue,
      }, {
        start: archiveRow.date, end: archiveRow.date, basis: `BOP-goods-${field}`, unit: 'US$ million',
        value: archiveRow[field], evidence: archiveRow.evidence?.[field],
      }));
    }
  }
  existing.countryReconciliation = reconciliation;
  await writeJson('trade.json', existing);

  console.log(`  📊 Top ${topExportCountries.length} export destinations (${exportCountryPeriod || 'period unknown'}), Top ${topImportCountries.length} import sources (${importCountryPeriod || 'period unknown'})`);
  console.log(`  🌍 Country monthly snapshot: ${countries.length} partners (latest ${countryMonthly.latestMonth || '?'}, ${fytdLabel})`);
  return { exports: topExportCountries.length, imports: topImportCountries.length };
}

// ═══════════════════════════════════════════════════
// DERIVED RESERVES ADEQUACY — official reserves / goods imports
// ═══════════════════════════════════════════════════

async function updateReservesAdequacyFromData() {
  console.log('\n🏦 Updating reserves adequacy from canonical official data...');

  const reserves = await readJson('reserves.json');
  const trade = await readJson('trade.json');
  const existing = await readJson('reserves-adequacy.json');
  const latest = reserves.weekly?.at(-1);
  const imports = (trade.monthly || []).slice(-12);

  if (!latest || imports.length !== 12) {
    throw new Error('Reserves adequacy requires the latest reserves point and 12 months of goods imports');
  }
  consecutiveMonths(imports);
  for (const row of imports) sourceNumber(row.imports, { context: `Reserve adequacy imports ${row.date}` });
  sourceNumber(latest.sbp, { context: 'Latest official SBP reserves' });
  sourceNumber(latest.banks, { context: 'Latest official bank reserves' });
  sourceNumber(latest.total, { context: 'Latest official total reserves' });

  const averageMonthlyImports = imports.reduce((sum, row) => sum + row.imports, 0) / imports.length;
  if (averageMonthlyImports <= 0) throw new Error('Reserves adequacy requires positive average imports');
  const importCoverMonths = Math.round((latest.sbp / averageMonthlyImports) * 10) / 10;
  const date = latest.date;
  const monthKey = date.slice(0, 7);
  const label = new Date(`${date}T00:00:00Z`).toLocaleDateString('en-US', {
    day: 'numeric',
    month: 'short',
    year: 'numeric',
    timeZone: 'UTC',
  });
  const trajectory = (existing.trajectory || [])
    .filter(point => point.date !== monthKey)
    .concat({
      date: monthKey,
      sbpReserves: round2(latest.sbp / 1000),
      importCoverMonths,
      label,
      status: latest.status ?? 'not-stated',
      evidence: {
        sbpReserves: latest.evidence?.sbp ? derivedEvidence([latest.evidence.sbp], 'US$ million divided by 1000') : null,
        importCoverMonths: latest.evidence?.sbp && imports.every(row => row.evidence?.imports)
          ? derivedEvidence([latest.evidence.sbp, ...imports.map(row => row.evidence.imports)], 'SBP reserves / average of 12 consecutive monthly BOP goods imports') : null,
      },
    })
    .sort((a, b) => a.date.localeCompare(b.date));

  const sbpReserves = round2(latest.sbp / 1000);
  const totalReserves = round2(latest.total / 1000);
  const bankReserves = round2(latest.banks / 1000);
  const averageImportsBn = round2(averageMonthlyImports / 1000);
  await writeJson('reserves-adequacy.json', {
    ...existing,
    current: {
      ...existing.current,
      sbpReserves,
      totalReserves,
      bankReserves,
      asOf: date,
      importCoverMonths,
      importCoverLabel: 'Goods-import cover',
      importCoverNote: `Computed from SBP-held reserves of $${sbpReserves}bn and the trailing 12-month average monthly goods-import bill of $${averageImportsBn}bn.`,
      importCoverBasis: 'SBP-held reserves divided by the trailing 12-month average of official SBP monthly goods imports.',
      status: latest.status ?? 'not-stated',
      evidence: {
        sbpReserves: latest.evidence?.sbp ? derivedEvidence([latest.evidence.sbp], 'US$ million divided by 1000') : null,
        totalReserves: latest.evidence?.total ? derivedEvidence([latest.evidence.total], 'US$ million divided by 1000') : null,
        bankReserves: latest.evidence?.banks ? derivedEvidence([latest.evidence.banks], 'US$ million divided by 1000') : null,
        importCoverMonths: latest.evidence?.sbp && imports.every(row => row.evidence?.imports)
          ? derivedEvidence([latest.evidence.sbp, ...imports.map(row => row.evidence.imports)], 'SBP reserves / average of 12 consecutive monthly BOP goods imports') : null,
      },
    },
    trajectory,
    context: `The latest official SBP reserve stock covers about ${importCoverMonths} months of the trailing goods-import bill on this dashboard's stated calculation. This is a transparent directional measure, not the IMF's broader reserve-adequacy metric.`,
    sourceUrl: existing.sourceUrl ?? null,
    lastVerified: existing.lastVerified ?? null,
    verifiedFrom: existing.verifiedFrom ?? [],
    methodologyNote: `The current ${importCoverMonths}-month figure is calculated as $${sbpReserves}bn of SBP-held reserves divided by $${averageImportsBn}bn, the trailing 12-month average of SBP monthly goods imports through ${imports.at(-1).date}. It therefore measures goods-import cover and may differ from official or IMF measures that use broader imports or reserve-adequacy methods. Historical trajectory points are reported estimates and are not recalculated on the same basis.`,
  });
  console.log(`  📊 ${importCoverMonths} months of trailing goods imports as of ${date}`);
}

// ═══════════════════════════════════════════════════
// KPI GENERATION — derives all KPIs from canonical data files
// ═══════════════════════════════════════════════════

async function generateKpiFromData({ readData = readJson, writeData = writeJson, recordCitations = recordKpiProvenance } = {}) {
  console.log('\n📊 Generating KPI summary from canonical data files...');

  const today = new Date().toISOString().split('T')[0];
  const indicators = [];
  const withheldIndicators = [];
  const isWithheld = id => withheldIndicators.some(indicator => indicator.id === id);
  const readInput = async (filename, id, requiredPaths = []) => {
    const data = await readData(filename);
    const publication = data.publication;
    if (publication && ['partial', 'withheld'].includes(publication.status)) {
      if (publication.policy !== 'official-only' || typeof publication.reason !== 'string' || !publication.reason.trim()
        || !Array.isArray(publication.withheldFields) || publication.withheldFields.some(path => typeof path !== 'string')
        || (publication.status === 'partial' && publication.withheldFields.length === 0)) {
        throw new Error(`Invalid withholding metadata for ${filename}`);
      }
      const blocksValue = publication.status === 'withheld' || requiredPaths.some(required =>
        publication.withheldFields.some(path => path === '' || path === required || required.startsWith(`${path}.`) || path.startsWith(`${required}.`)));
      if (blocksValue) {
        withheldIndicators.push({ id, dataset: filename, reason: publication.reason });
        console.log(`  ⏭  ${id} withheld: ${publication.reason}`);
      }
    }
    return data;
  };

  // --- Reserves (from reserves.json) ---
  try {
    const reserves = await readInput('reserves.json', 'reserves');
    const pts = reserves.weekly || [];
    if (pts.length > 0 && !isWithheld('reserves')) {
      const latest = pts[pts.length - 1];
      const prev = pts.at(-2) ?? null;
      const sbpBn = round2(sourceNumber(latest.sbp) / 1000);
      const totalBn = round2(sourceNumber(latest.total) / 1000);
      const changeBn = prev ? round2((latest.total - sourceNumber(prev.total)) / 1000) : null;
      const trend = changeBn === null ? 'unavailable' : Math.abs(changeBn) < 0.3 ? 'stable' : changeBn > 0 ? 'up' : 'down';
      // Format date for display
      const dateLabel = latest.date.length > 7 ? latest.date : latest.date; // weekly dates are full, monthly are YYYY-MM
      indicators.push({
        id: 'reserves', label: 'Foreign Reserves (Total)',
        value: totalBn, unit: '$ Billion', period: dateLabel,
        change: changeBn, changeUnit: '$B', changeBasis: prev ? `vs week ending ${prev.date}` : 'no prior observation',
        trend, sentiment: directionalSentiment(trend), source: 'SBP',
        provenanceKey: 'reserves.weekly.total',
        status: latest.status ?? 'not-stated', evidence: latest.evidence?.total ? { ...latest.evidence.total, scale: 0.001 } : null,
        sub: `SBP: $${sbpBn}B · Banks: $${round2(latest.banks / 1000)}B`,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate reserves KPI: ${err.message}`, { cause: err });
  }

  // --- Exchange Rate (from exchange-rates.json) ---
  try {
    const ex = await readInput('exchange-rates.json', 'exchange-rate');
    const pts = ex.monthly || [];
    if (pts.length > 0 && !isWithheld('exchange-rate')) {
      const latest = pts[pts.length - 1];
      const prev = pts.at(-2) ?? null;
      sourceNumber(latest.USD);
      const change = prev ? round2(latest.USD - sourceNumber(prev.USD)) : null;
      const trend = change === null ? 'unavailable' : Math.abs(change) < 1 ? 'stable' : change > 0 ? 'up' : 'down';
      indicators.push({
        id: 'exchange-rate', label: 'PKR / USD',
        value: latest.USD, unit: 'PKR', period: latest.date,
        change, changeUnit: 'PKR', changeBasis: prev ? `vs ${prev.date}` : 'no prior observation',
        trend, sentiment: directionalSentiment(trend, false), source: 'SBP',
        provenanceKey: 'exchange-rates.monthly.usd',
        status: latest.status ?? 'not-stated', evidence: latest.evidence?.USD ?? null,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate exchange-rate KPI: ${err.message}`, { cause: err });
  }

  // --- Remittances (from remittances.json) ---
  try {
    const rem = await readInput('remittances.json', 'remittances');
    const pts = rem.monthly || [];
    if (pts.length > 0 && !isWithheld('remittances')) {
      const latest = pts[pts.length - 1];
      const prev = pts.at(-2) ?? null;
      const valBn = latest.total === null ? null : round2(sourceNumber(latest.total) / 1000);
      const changeBn = latest.total !== null && typeof prev?.total === 'number' ? round2((latest.total - prev.total) / 1000) : null;
      const trend = changeBn === null ? 'unavailable' : Math.abs(changeBn) < 0.2 ? 'stable' : changeBn > 0 ? 'up' : 'down';
      indicators.push({
        id: 'remittances', label: 'Remittances (Monthly)',
        value: valBn, unit: '$ Billion', period: latest.date,
        change: changeBn, changeUnit: '$B', changeBasis: prev ? `vs ${prev.date}` : 'no prior observation',
        trend, sentiment: directionalSentiment(trend), source: 'SBP',
        provenanceKey: 'remittances.monthly.total',
        status: latest.observations?.total?.status ?? 'not-stated',
        evidence: latest.evidence?.total ? { ...latest.evidence.total, scale: 0.001 } : null,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate remittances KPI: ${err.message}`, { cause: err });
  }

  // --- FDI (from fdi.json) ---
  try {
    const fdi = await readInput('fdi.json', 'fdi');
    if (fdi.fytdComparison && !isWithheld('fdi')) {
      const cur = fdi.fytdComparison.current;
      const prior = fdi.fytdComparison.prior;
      const curNet = cur.net_fdi ?? cur.net;
      const priorNet = prior?.net_fdi ?? prior?.net ?? null;
      const changePct = typeof curNet === 'number' && typeof priorNet === 'number' && priorNet !== 0 ? round2(((curNet - priorNet) / Math.abs(priorNet)) * 100) : null;
      const netBn = curNet === null ? null : round2(sourceNumber(curNet) / 1000);
      const trend = changePct === null ? 'unavailable' : changePct > 2 ? 'up' : changePct < -2 ? 'down' : 'stable';
      indicators.push({
        id: 'fdi', label: 'Net FDI',
        value: netBn, unit: '$B', decimals: 2,
        period: `${fdi.fytdComparison.period} ${cur.label}${cur.status === 'provisional' ? ' (P)' : ''}`,
        change: changePct, changeUnit: '%', changeBasis: prior ? `vs ${fdi.fytdComparison.period} ${prior.label}` : 'no prior observation',
        trend, sentiment: directionalSentiment(trend), source: 'SBP',
        provenanceKey: 'fdi.fytd.current',
        status: cur.status ?? 'not-stated', evidence: cur.evidence?.net_fdi ? { ...cur.evidence.net_fdi, scale: 0.001 } : null,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate FDI KPI: ${err.message}`, { cause: err });
  }

  // --- IT & Services (from services.json) ---
  try {
    const svc = await readInput('services.json', 'it_exports');
    if ((svc.itHeadline || svc.comparison) && !isWithheld('it_exports')) {
      const current = svc.itHeadline ? svc.itHeadline.fytd : svc.comparison.current?.itCredit;
      const prior = svc.itHeadline ? svc.itHeadline.fytdPrior : svc.comparison.prior?.itCredit;
      const periodLabel = svc.itHeadline?.fytdLabel || `${svc.comparison.period} ${svc.comparison.currentLabel}`;
      fiscalWindow(periodLabel, svc.itHeadline?.fiscalYear ?? svc.comparison.current?.fy);
      const itBn = current === null ? null : round2(sourceNumber(current) / 1000);
      const priorBn = prior === null || prior === undefined ? null : round2(sourceNumber(prior) / 1000);
      const changePct = itBn !== null && priorBn !== null && prior !== 0 ? round2(((current - prior) / prior) * 100) : null;
      const trend = changePct === null ? 'unavailable' : changePct === 0 ? 'stable' : changePct > 0 ? 'up' : 'down';
      indicators.push({
        id: 'it_exports', label: 'IT & Telecom Exports',
        value: itBn, unit: '$B', decimals: 2,
        period: periodLabel,
        change: changePct, changeUnit: '%',
        changeBasis: svc.itHeadline?.fytdPriorLabel
          ? `vs ${svc.itHeadline.fytdPriorLabel}`
          : `vs ${svc.comparison.period} ${svc.comparison.priorLabel}`,
        trend,
        sentiment: directionalSentiment(trend),
        source: 'SBP',
        provenanceKey: svc.itHeadline ? 'services.itHeadline.fytd' : 'services.itTelecom.credit',
        status: svc.itHeadline?.statuses?.fytd ?? svc.comparison.current?.status ?? 'not-stated',
        evidence: (svc.itHeadline?.evidence?.fytd ?? svc.comparison.current?.evidence?.itCredit)
          ? derivedEvidence([svc.itHeadline?.evidence?.fytd ?? svc.comparison.current?.evidence?.itCredit], 'US$ million divided by 1000 to US$ billion') : null,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate IT exports KPI: ${err.message}`, { cause: err });
  }

  // --- GDP Growth (from fiscal.json) ---
  try {
    const fiscal = await readInput('fiscal.json', 'gdp-growth');
    const annual = fiscal.annual || [];
    if (annual.length > 0 && !isWithheld('gdp-growth')) {
      const latest = annual[annual.length - 1];
      const prev = annual.length > 1 ? annual[annual.length - 2] : null;
      sourceNumber(latest.gdpGrowth);
      const change = typeof prev?.gdpGrowth === 'number' ? round2(latest.gdpGrowth - prev.gdpGrowth) : null;
      const trend = change === null ? 'unavailable' : change > 0.2 ? 'up' : change < -0.2 ? 'down' : 'stable';
      indicators.push({
        id: 'gdp-growth', label: 'GDP Growth Rate',
        value: latest.gdpGrowth, unit: '%',
        period: latest.year,
        change, changeUnit: 'pp', changeBasis: prev ? `vs ${prev.year}` : 'no prior year',
        trend, sentiment: directionalSentiment(trend), source: 'PBS via SBP',
        provenanceKey: 'fiscal.gdpGrowth.latest',
        status: latest.gdpGrowthStatus ?? 'not-stated', evidence: latest.evidence?.gdpGrowth ?? null,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate GDP KPI: ${err.message}`, { cause: err });
  }

  // --- Inflation (from inflation.json) ---
  try {
    const inf = await readInput('inflation.json', 'inflation');
    const cpi = inf.national_cpi?.data || [];
    if (cpi.length > 0 && !isWithheld('inflation')) {
      const latest = cpi[cpi.length - 1];
      const prev = cpi.at(-2) ?? null;
      const change = typeof latest.value === 'number' && typeof prev?.value === 'number' ? round2(latest.value - prev.value) : null;
      const trend = change === null ? 'unavailable' : latest.value > 5 ? (change > 0.5 ? 'up' : change < -0.5 ? 'down' : 'stable') : 'down';
      indicators.push({
        id: 'inflation', label: 'CPI Inflation (YoY)',
        value: round2(latest.value), unit: '%',
        period: latest.date,
        change, changeUnit: 'pp', changeBasis: prev ? `vs ${prev.date}` : 'no prior observation',
        trend,
        sentiment: latest.value === null ? 'neutral' : targetBandSentiment(latest.value, trend, 5, 7),
        source: 'PBS',
        provenanceKey: 'inflation.nationalCpi.latest',
        status: latest.status ?? 'not-stated', evidence: latest.evidence ?? null,
        sub: `SBP target: 5–7%`,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate inflation KPI: ${err.message}`, { cause: err });
  }

  // --- FBR Tax Collection (from fbr-tax.json) ---
  try {
    const fbr = await readInput('fbr-tax.json', 'fbr-tax', ['fytd.net']);
    const pts = (fbr.monthly || []).slice().sort((a, b) => a.date.localeCompare(b.date));
    const latest = pts[pts.length - 1];
    if (fbr.fytd && latest && !isWithheld('fbr-tax')) {
      const f = fbr.fytd;
      if (f.sourceType && f.sourceType !== 'official-primary') throw new Error('FBR KPI requires an official primary source');
      const net = sourceNumber(f.net);
      const priorNet = sourceNumber(f.priorNet, { required: false });
      const target = sourceNumber(f.target, { required: false });
      const latestNet = sourceNumber(latest.net, { required: false });
      const growthPct = priorNet !== null && priorNet !== 0 ? round2(((net - priorNet) / priorNet) * 100) : null;
      const trend = growthPct === null ? 'unavailable' : growthPct > 0.5 ? 'up' : growthPct < -0.5 ? 'down' : 'stable';
      const gap = target !== null ? Math.round(net - target) : null;
      const sub = gap != null
        ? `Target ₨${(target / 1000).toFixed(2)}T · ${gap >= 0 ? 'surplus' : 'shortfall'} ₨${Math.abs(gap)}B`
        : latestNet === null ? 'Latest monthly collection unavailable' : `Latest: ₨${Math.round(latestNet)}B (${latest.date})`;
      indicators.push({
        id: 'fbr-tax', label: fbrCollectionLabel(f),
        value: round2(net / 1000), unit: 'T PKR',
        period: f.period,
        change: growthPct, changeUnit: '%',
        changeBasis: f.priorPeriod ? `vs ${f.priorPeriod}` : 'vs same period last year',
        trend,
        sentiment: gap == null ? directionalSentiment(trend) : gap >= 0 ? 'positive' : 'negative',
        source: f.sourceType === 'secondary-attributed'
          ? (f.sourceLabel || 'Secondary report citing provisional FBR data')
          : 'FBR',
        sourceType: f.sourceType || 'official-primary',
        provenanceKey: 'fbr.fytd.net',
        status: f.status ?? 'not-stated', evidence: f.evidence?.net ?? f.evidence ?? null,
        sub,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate FBR KPI: ${err.message}`, { cause: err });
  }

  // --- Policy Rate (from monetary-policy.json) ---
  // Derived from the SBP Monetary Policy Committee decision series — never
  // copied forward from the previous kpi-summary.json, which would let a stale
  // rate persist indefinitely while `lastUpdated` kept advancing.
  try {
    const mp = await readInput('monetary-policy.json', 'policy-rate', ['currentRate']);
    if (!isWithheld('policy-rate')) {
      const decisions = (mp.decisions || []).slice().sort((a, b) => a.date.localeCompare(b.date));
      const lastDecision = decisions[decisions.length - 1];
      const currentRate = sourceNumber(mp.currentRate, { context: 'Policy rate' });
      if (lastDecision && Math.abs(sourceNumber(lastDecision.rate) - currentRate) > 0.001) {
        throw new Error(
          `monetary-policy.json currentRate (${mp.currentRate}%) disagrees with the latest ` +
          `decision on ${lastDecision.date} (${lastDecision.rate}%)`,
        );
      }
      const changePp = lastDecision?.changeBps != null ? round2(sourceNumber(lastDecision.changeBps) / 100) : null;
      const trend = changePp === null ? 'unavailable' : changePp > 0 ? 'up' : changePp < 0 ? 'down' : 'stable';
      indicators.push({
        id: 'policy-rate',
        label: 'SBP Policy Rate',
        value: currentRate,
        unit: '%',
        period: mp.asOf || lastDecision?.date || null,
        change: changePp,
        changeUnit: 'pp',
        changeBasis: lastDecision ? `last MPC decision ${lastDecision.date}` : 'no decision on record',
        trend,
        sentiment: 'neutral',
        source: 'SBP Monetary Policy Committee',
        provenanceKey: 'monetaryPolicy.currentRate',
        status: mp.status ?? lastDecision?.status ?? 'not-stated', evidence: mp.evidence?.currentRate ?? lastDecision?.evidence?.rate ?? null,
        sub: mp.nextMeeting
          ? `Next MPC: ${mp.nextMeeting.dateText || mp.nextMeeting.date || 'date to be announced'}`
          : undefined,
      });
    }
  } catch (err) {
    throw new Error(`Could not generate policy-rate KPI: ${err.message}`, { cause: err });
  }

  const expectedIds = ['reserves', 'exchange-rate', 'remittances', 'fdi', 'it_exports', 'gdp-growth', 'inflation', 'fbr-tax', 'policy-rate'];
  const missingIds = expectedIds.filter(id => !indicators.some(indicator => indicator.id === id) && !isWithheld(id));
  if (missingIds.length > 0) {
    throw new Error(`KPI generation incomplete; missing: ${missingIds.join(', ')}`);
  }
  // A bare "+1.4" next to a headline number is unreadable and, worse, invites a
  // reader to compare a $B week-on-week delta with a percent YoY change. Every
  // KPI must therefore declare what its `change` is measured in and against what.
  const unlabelled = indicators.filter(i => i.change != null && (!i.changeUnit || !i.changeBasis));
  if (unlabelled.length > 0) {
    throw new Error(
      `KPI change values must declare changeUnit and changeBasis; missing on: ${unlabelled.map(i => i.id).join(', ')}`,
    );
  }
  const kpi = { lastUpdated: today, indicators, ...(withheldIndicators.length ? { withheldIndicators } : {}) };
  await writeData('kpi-summary.json', kpi);
  await recordCitations(indicators);
  console.log(`  ✅ Generated ${indicators.length} KPI indicators from data files`);
}

// Record where each headline KPI came from so the UI can cite it precisely.
async function recordKpiProvenance(indicators) {
  const byId = Object.fromEntries(indicators.map((i) => [i.id, i]));
  const byKey = Object.fromEntries(indicators.map(indicator => [indicator.provenanceKey, indicator]));
  const cite = async (key, entry) => {
    const indicator = byKey[key];
    const evidence = indicator?.evidence ?? entry.evidence;
    if (!evidence) return;
    const direct = evidence.artifactId ? evidence : evidence.inputs?.find(input => input.artifactId);
    if (!direct) throw new Error(`Cannot record ${key} without source artifact inputs`);
    return recordProvenance(key, {
      ...entry, evidence, status: indicator?.status ?? entry.status ?? 'not-stated',
      sourceKey: direct.sourceKey,
      sheet: direct.locator?.sheet ?? entry.sheet ?? null,
      location: JSON.stringify(evidence.locator ?? evidence),
      retrievedAt: direct.retrievedAt,
    });
  };

  if (byId.reserves) {
    await cite('reserves.weekly.total', {
      label: 'Total liquid foreign exchange reserves (SBP + scheduled banks)',
      sourceKey: 'forex.pdf',
      location: 'Weekly reserves statement, "Total Liquid Foreign Reserves" line',
      period: byId.reserves.period,
      unit: 'US$ billion',
      value: byId.reserves.value,
    });
  }
  if (byId['exchange-rate']) {
    await cite('exchange-rates.monthly.usd', {
      label: 'PKR/USD interbank closing rate, monthly average',
      sourceKey: 'IBF_Arch.xls',
      location: 'Interbank rates archive, USD column',
      period: byId['exchange-rate'].period,
      unit: 'PKR per USD',
      value: byId['exchange-rate'].value,
    });
  }
  if (byId.remittances) {
    await cite('remittances.monthly.total', {
      label: 'Workers\u2019 remittances, monthly total',
      sourceKey: 'sbp-easydata',
      location: 'Workers\u2019 remittances by country of origin, total',
      period: byId.remittances.period,
      unit: 'US$ billion',
      value: byId.remittances.value,
    });
  }
  if (byId.fdi) {
    const fdiJson = await readJson('fdi.json');
    const fytd = fdiJson.fytdComparison || {};
    await cite('fdi.fytd.current', {
      label: 'Net foreign direct investment, fiscal year to date',
      sourceKey: fytd.sourceFile || 'NetinflowSummary.xls',
      sheet: fytd.sourceSheet || 'Summary',
      location: fytd.sourceLocation || '"Direct Investment" row, current fiscal-year-to-date column',
      period: byId.fdi.period,
      status: byId.fdi.status,
      unit: 'US$ billion',
      value: byId.fdi.value,
    });
  }
  if (byId.it_exports) {
    const servicesJson = await readJson('services.json');
    const headline = servicesJson.itHeadline;
    await cite(byId.it_exports.provenanceKey, {
      label: 'Telecommunications, computer and information services exports (credit)',
      sourceKey: headline ? 'ExportsImports-Goods.pdf' : 'dt.xls',
      location: headline
        ? 'IT & Telecommunication Services, current fiscal-year cumulative column'
        : 'EBOPS line 9 "Telecommunications, Computer and information services", credit column',
      period: byId.it_exports.period,
      status: byId.it_exports.status,
      unit: 'US$ billion',
      value: byId.it_exports.value,
    });
  }
  if (byId['gdp-growth']) {
    await cite('fiscal.gdpGrowth.latest', {
      label: 'Real GDP growth rate at constant basic prices of 2015-16',
      sourceKey: 'GDP_table.xlsx',
      sheet: 'Annual',
      location: '"GDP Growth Rate (%)" row',
      period: byId['gdp-growth'].period,
      status: byId['gdp-growth'].status,
      unit: '%',
      value: byId['gdp-growth'].value,
    });
  }
  if (byId.inflation) {
    await cite('inflation.nationalCpi.latest', {
      label: 'National Consumer Price Index, year-on-year inflation',
      sourceKey: 'pbs-cpi',
      location: 'Monthly CPI review, national year-on-year rate',
      period: byId.inflation.period,
      unit: '%',
      value: byId.inflation.value,
    });
  }
  if (byId['fbr-tax']) {
    await cite('fbr.fytd.net', {
      label: 'FBR net tax collection, fiscal year to date',
      sourceKey: 'fbr-collection',
      location: 'FBR revenue collection statement, net collection',
      period: byId['fbr-tax'].period,
      status: byId['fbr-tax'].status,
      unit: 'PKR trillion',
      value: byId['fbr-tax'].value,
      note: byId['fbr-tax'].sourceType === 'secondary-attributed'
        ? 'Currently sourced from press reporting of provisional FBR figures, not an FBR publication.'
        : null,
    });
  }
  if (byId['policy-rate']) {
    await cite('monetaryPolicy.currentRate', {
      label: 'SBP policy (target) rate set by the Monetary Policy Committee',
      sourceKey: 'sbp-monetary-policy',
      location: 'Latest Monetary Policy Statement',
      period: byId['policy-rate'].period,
      unit: '%',
      value: byId['policy-rate'].value,
    });
  }

  // Figures that are cited on charts but are not headline KPIs.
  try {
    const trade = await readJson('trade.json');
    const latest = (trade.monthly || []).at(-1);
    if (latest) {
      await cite('trade.monthly.balance', {
        label: 'Monthly goods trade balance (BOP basis)',
        sourceKey: 'exp_import_BOP.xls',
        location: 'Exports and imports of goods, monthly',
        period: latest.date,
        unit: 'US$ million',
        value: latest.balance,
        status: latest.status ?? 'not-stated', evidence: latest.evidence?.balance ?? null,
        derivedFrom: `Exports of $${latest.exports}M less imports of $${latest.imports}M as published by SBP`,
      });
    }
  } catch (err) {
    throw new Error(`Could not record trade provenance: ${err.message}`, { cause: err });
  }

  const count = await flushProvenance();
  console.log(`  ✅ provenance.json now cites ${count} figures`);
}

async function main() {
  console.log('\n🇵🇰 Parsing SBP Excel files...\n');
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  console.log('  Source: scripts/sbp-raw/');
  console.log('  Target: public/data/');
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');

  const summary = {};

  // 1. Try downloading exchange rate archive first
  const archiveOk = await downloadExchangeRateArchive();

  // 2. Parse trade data
  summary.trade = await updateTrade();

  // 3. Parse FDI data
  summary.fdi = await updateFdi();

  // 4. Parse GDP / fiscal data
  summary.fiscal = await updateGdpFiscal();

  // 5. (Balance of Payments diagnostics removed — see note above.)

  // 6. Exchange rates (if archive was downloaded)
  if (archiveOk) {
    summary.exchangeRates = await updateExchangeRates();
  }

  // 7. Reserves (forex.pdf)
  summary.reserves = await updateReserves();

  // 8. Services (dt.xls — EBOPS)
  summary.services = await updateServices();

  // 8b. Services headline from the BOP summary. Runs after updateServices()
  // because it augments the same file, and SBP refreshes it a month earlier.
  summary.bopServices = await updateBopServices();

  // 8c. Newer IT headline from SBP's monthly goods-and-services table. This
  // advances the aggregate without relabelling lagging EBOPS subcomponents.
  summary.servicesHeadline = await updateServicesHeadline();

  // 9. Trade by country
  summary.tradeCountries = await updateTradeCountries();

  // 10. Refresh derived metrics from the newly parsed canonical data.
  await updateReservesAdequacyFromData();

  // 11. Regenerate KPI summary from all canonical data files
  await generateKpiFromData();

  // ─── Summary ───
  console.log('\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  console.log('📋 Parse Summary');
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  console.log(`  ✅ trade.json       — ${summary.trade} monthly data points`);
  console.log(`  ✅ fdi.json         — ${summary.fdi?.sectors} sectors, ${summary.fdi?.countries} countries, ${summary.fdi?.years} FY`);
  console.log(`  ✅ fiscal.json      — ${summary.fiscal} fiscal years`);
  console.log(`  ✅ kpi-summary.json — derived from canonical data files`);
  if (summary.exchangeRates) {
    console.log(`  ✅ exchange-rates.json — ${summary.exchangeRates} months`);
  } else {
    console.log(`  ⏭  exchange-rates.json — no archive data available`);
  }
  if (summary.reserves) {
    console.log(`  ✅ reserves.json    — ${summary.reserves} data points`);
  }
  if (summary.services) {
    console.log(`  ✅ services.json    — updated from EBOPS data`);
  }
  if (summary.tradeCountries) {
    console.log(`  ✅ trade.json       — added ${summary.tradeCountries.exports} export + ${summary.tradeCountries.imports} import countries`);
  }
  console.log('\n✨ Done!\n');
}
